def sha(data: bytes) -> str:
    """Bind exact bytes, retaining raw and rendered identities separately."""
    return hashlib.sha256(data).hexdigest()


def resolve(path: str) -> Path:
    """Resolve only the two previously defined evidence namespaces."""
    return WORKER / path[8:] if path.startswith("SCRATCH/") else ROOT / path


def bind(reference: dict) -> bytes:
    """Verify an existing digest before interpreting its evidence."""
    raw = resolve(reference["path"]).read_bytes()
    assert sha(raw) == reference["sha256"], reference["path"]
    checked[reference["path"]] = reference["sha256"]
    return raw


def portable(text: str) -> str:
    """Reproduce the worker's declared path substitution without changing raw data."""
    for path, label in ((ROOT, "WORKSPACE"), (WORKER, "SCRATCH")):
        text = re.sub(re.escape(str(path)), label, text, flags=re.IGNORECASE)
        text = re.sub(re.escape(path.as_posix()), label, text, flags=re.IGNORECASE)
    return text


def git(*args: str) -> bytes:
    """Inspect existing Git objects without staging or modifying the checkout."""
    return subprocess.check_output(["git", *args])


def ac_state(path: Path) -> dict:
    """Count only authoritative acceptance criteria, preserving the entire file."""
    raw = path.read_bytes()
    section = re.search(r"(?ms)^## Acceptance Criteria[^\n]*\n(.*?)(?=^## |\Z)", raw.decode()).group(1)
    states = re.findall(r"(?m)^- \[([ x])\] ", section)
    assert len(states) == 66 and states.count("x") == 13, str(path)
    return {"path": path.as_posix(), "sha256": sha(raw), "passed": 13, "total": 66}
