"""Prepare LF-only P4-T7 runners from retained P4-T6 mechanics; execute no gate."""
from pathlib import Path

scratch = Path(__file__).parent

def save(name, text):
    path = scratch / name
    assert not path.exists(), name
    path.write_bytes((text.replace('\r\n', '\n').rstrip() + '\n').encode())

architecture = (scratch / 'issue-980-p4t6-final-architecture.py').read_text('utf-8').replace('P4-T6', 'P4-T7').replace('== 7', '== 14')
save('issue-980-p4t7-final-architecture.py', architecture)
gates = (scratch / 'issue-980-p4t6-final-gates.py').read_text('utf-8').replace('P4-T6', 'P4-T7').replace('p4t6', 'p4t7')
gates = gates.replace("len(manifest['subset_manifests']) == 2", "len(manifest['subset_manifests']) == 5").replace('Both frozen subsets', 'All five frozen subsets')
gates = gates.replace('len(code) == 7', 'len(code) == 14')
start = gates.index("assert {Path(path).name for path in code}")
end = gates.index('\nreceipts = []', start)
gates = gates[:start] + "assert {Path(path).name for path in code} == {'dependency_inputs.py', 'dependency_values.py', 'dependency_node.py', 'dependency_python.py', 'dependency_dotnet.py', 'dependencies.py', 'test_dependency_inputs.py', 'test_dependency_input_limits.py', 'test_dependency_node.py', 'test_dependency_node_limits.py', 'test_dependency_python.py', 'test_dependency_dotnet.py', 'test_dependencies.py', 'test_dependency_projection_limits.py'}" + gates[end:]
start = gates.index('batches = (')
end = gates.index('\nfor batch, module_names', start)
gates = gates[:start] + '''batches = (
    ('shared', ('dependency_input_limits', 'dependency_inputs'), ('dependency_inputs', 'dependency_values')),
    ('node', ('dependency_node_limits', 'dependency_node'), ('dependency_node',)),
    ('python', ('dependency_python',), ('dependency_python',)),
    ('dotnet', ('dependency_dotnet',), ('dependency_dotnet',)),
    ('integration', ('dependency_projection_limits', 'dependencies'), ('dependencies',)),
)''' + gates[end:]
gates = gates.replace("assert len(batch_nodes) == manifest['expected_test_counts'][batch] and len(batch_nodes) == len(set(batch_nodes)), (batch, len(batch_nodes))", "assert len(batch_nodes) == manifest['expected_test_counts'][batch] and len(batch_nodes) == len(set(batch_nodes)), (batch, len(batch_nodes))\n    assert set(batch_nodes) == set(manifest['prior_node_order'][batch]), ('case population changed', batch)")
gates = gates.replace('runtime callback/test', 'package install/restore/execution')
save('issue-980-p4t7-final-gates.py', gates)
