def check(name, condition):
    """Record an explicit assertion; failure terminates without a success result."""
    if not condition:
        raise AssertionError(name)
    checks.append(name)
    print('PASS ' + name, flush=True)


def put(path, content):
    """Create synthetic setup bytes only, before the read-only checkpoint."""
    destination = root / path
    assert root in destination.resolve().parents
    destination.parent.mkdir(parents=True, exist_ok=True)
    destination.write_bytes(content)


def tree():
    """Hash every fixture file, including Git internals, without changing it."""
    return {path.relative_to(root).as_posix(): hashlib.sha256(path.read_bytes()).hexdigest()
            for path in sorted(root.rglob('*')) if path.is_file()}


def git(*arguments):
    """Run explicit fixture-only Git operations with optional index writes disabled."""
    return subprocess.run(['git', '--no-optional-locks', '-C', str(root), *arguments],
                          capture_output=True, check=True).stdout


def run_git(argv, *, output_limit):
    """Provide the accepted bounded Git-reader seam; never evaluate shell text."""
    process = subprocess.Popen(argv, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    try:
        content = process.stdout.read(output_limit + 1)
        if len(content) > output_limit:
            process.kill()
            raise OSError('Native Git output exceeded its bounded read')
        if process.wait() != 0:
            raise OSError('Native Git read failed')
        return content
    finally:
        if process.poll() is None:
            process.kill()
        process.wait()
        process.stdout.close()


def immutable(value):
    """Parse synthetic detached JSON through the public immutable-value boundary."""
    parsed = parse_document(json.dumps(value), format='json')
    assert parsed.ok and isinstance(parsed.value, JsonObject)
    return parsed.value


def capture():
    """Use actual workspace loading and retained pinned source bytes."""

        metadata=primitives.metadata, read_bytes=primitives.read_bytes, catalog=catalog, repository=repository)
    assert workspace.snapshot.ok, [(item.code, item.path) for item in workspace.snapshot.findings]
    return pin_inventory_sources(workspace, repository)


def variant(manifest_changes=None, member_changes=None, *, seal=True):
    """Make an immutable adversarial capture; never write or alter loaded artifacts."""
    manifest = json.loads(snapshot.file(MANIFEST).content)
    member = json.loads(snapshot.file(MEMBER).content)
    if member_changes:
        member_changes(member)
    member_bytes = canonical_json(immutable(member))
    manifest['result_files'][0].update(sha256=hashlib.sha256(member_bytes).hexdigest(), size=len(member_bytes))
    if manifest_changes:
        manifest_changes(manifest)
    if seal:
        manifest['manifest_sha256'] = extension_manifest_digest(immutable(manifest))
    changes = {MANIFEST: canonical_json(immutable(manifest)), MEMBER: member_bytes}
    return replace(snapshot, manifest=replace(snapshot.manifest,
        files=tuple(replace(item, content=changes[item.path]) if item.path in changes else item
                    for item in snapshot.manifest.files)))


def reject(name, changed, expected, *, merge=False):
    """Require a fixed error from raw retained-byte reconstruction, without effects."""
    observed = None
    try:
        if merge:
            merge_extension_inventory(changed, timestamp=STAMP, author_or_agent=AUTHOR)
        else:
            validate_extension_results(changed)
    except ExtensionResultError as error:
        observed = error.code
        assert 'PRIVATE_VALUE' not in str(error)
    check(name, observed == expected)
