def read(path: Path) -> dict:
    """Decode trusted evidence with explicit UTF-8; do not execute source."""
    return json.loads(path.read_text(encoding="utf-8"))


def digest(raw: bytes) -> str:
    """Return the exact byte stream SHA-256."""
    return hashlib.sha256(raw).hexdigest()


def binding(path: Path) -> dict:
    """Bind a repository stream without copying its content."""
    raw = path.read_bytes()
    return {"path": path.relative_to(ROOT).as_posix(), "sha256": digest(raw),
            "bytes": len(raw), "physical_lines": len(raw.splitlines())}


def objects(value: object):
    """Enumerate evidence objects for exact recorded identity lookup."""
    if isinstance(value, dict):
        yield value
        for child in value.values():
            yield from objects(child)
    elif isinstance(value, list):
        for child in value:
            yield from objects(child)


def write(path: Path, value: object) -> None:
    """Write an authored LF-only JSON view with one terminal newline."""
    path.write_bytes((json.dumps(value, indent=2) + "\n").encode("utf-8"))
