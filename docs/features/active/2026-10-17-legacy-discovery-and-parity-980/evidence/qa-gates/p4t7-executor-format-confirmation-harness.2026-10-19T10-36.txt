import hashlib
import json
import multiprocessing

from pathlib import Path

ROOT = Path.cwd()
FEATURE = Path('docs/features/active/2026-10-17-legacy-discovery-and-parity-980')
REG = FEATURE / 'evidence/regression-testing'
QA = FEATURE / 'evidence/qa-gates'
R = 'packages/discovery-runtime/src/drm_copilot_discovery/analyzer/'


def binding(path, expected=None):
    raw = (ROOT / path).read_bytes()
    digest = hashlib.sha256(raw).hexdigest()
    if expected is not None:
        assert digest == expected, str(path)
    return {'path': str(path).replace('\\', '/'), 'sha256': digest, 'bytes': len(raw)}


def read(path, expected=None):
    item = binding(path, expected)
    return json.loads((ROOT / path).read_bytes()), item


def ini_child(connection, width, valid):
    from drm_copilot_discovery.analyzer.dependency_inputs import DependencyInputError, DependencyLimits
    from drm_copilot_discovery.analyzer.dependency_python import _IniParser, parse_python_dependencies
    text = '[metadata]\nx' + ' ' * width + ('!=value' if valid else '!')
    parser = _IniParser(interpolation=None, strict=True, empty_lines_in_values=False)
    assert parser._optcre is parser.OPTCRE
    assert parser.OPTCRE.pattern == r'(?P<option>[^=]+)(?P<vi>=)(?P<value>.*)$'
    started = time.perf_counter()
    try:
        result = parse_python_dependencies(text.encode(), format='setup-cfg', limits=DependencyLimits())
    except DependencyInputError as error:
        observed = {'status': 'fixed-error', 'code': error.code}
    else:
        observed = {'status': 'returned', 'candidate_count': len(result.candidates), 'consumed_nodes': result.consumed_nodes, 'consumed_text_bytes': result.consumed_text_bytes}
    connection.send({**observed, 'input_bytes': len(text.encode()), 'parser_seconds': time.perf_counter() - started, 'fixed_matcher_identity': True})
    connection.close()

                assert value['sha256'] == sources[path]

            gates.append(bound_gate)

        assert summary['num_statements'] == summary['covered_lines'] == lines
        assert summary['num_branches'] == summary['covered_branches'] == branches

        coverage.append({**item, 'lines': lines, 'branches': branches, 'named_class_callable_ranges': len(units)})

                process.terminate()
                process.join(timeout=2)

            observation = parent.recv()
            parent.close()
            assert observation['status'] == ('returned' if valid else 'fixed-error')

                assert observation['candidate_count'] == 0
