def digest(raw: bytes) -> str:
    """Return exact byte identity."""
    return hashlib.sha256(raw).hexdigest()


def read(path: Path) -> dict:
    """Decode retained JSON using explicit UTF-8."""
    return json.loads(path.read_text(encoding="utf-8"))


def reference(path: Path) -> dict:
    """Describe an existing canonical stream."""
    raw = path.read_bytes()
    return {"path": path.relative_to(ROOT).as_posix(), "sha256": digest(raw), "bytes": len(raw)}


def write(path: Path, value: object) -> None:
    """Write authored LF-terminated JSON without changing raw artifacts."""
    path.write_bytes((json.dumps(value, indent=2) + "\n").encode("utf-8"))


def one(stage: str, prefix: str = PREFIX) -> Path:
    """Resolve the exact current attempt's singular stage receipt."""
    paths = list(OUT.glob(prefix + "-" + stage + ".*.json"))
    assert len(paths) == 1, paths
    return paths[0]

        (TESTS[2], "test_missing_reference_revision_rejects_before_output_reservation"),

        lines = raw.decode("utf-8").splitlines(keepends=True)
        node = next(n for n in ast.parse("".join(lines)).body
                    if isinstance(n, ast.FunctionDef) and n.name == new_test)
        start = min([node.lineno, *(n.lineno for n in node.decorator_list)])
        text = "".join(lines[:start - 1]).rstrip("\r\n") + "\n"
        if path == TESTS[1]:
            text = text.replace("from drm_copilot_discovery import acceptance_generation as generation_module\n", "", 1)
            text = text.replace("from drm_copilot_discovery.acceptance_generation_context import (\n    GenerationContext,\n    generation_preflight,\n)\n", "", 1)
            text = text.replace("    from typing import Protocol\n\n", "", 1)
            text = text.replace("    from drm_copilot_discovery.references import ArtifactRecord\n", "", 1)
            protocol = next(n for n in ast.walk(ast.parse(text))
                            if isinstance(n, ast.ClassDef) and n.name == "ScenarioSelection")
            lines = text.splitlines(keepends=True)
            text = ("".join(lines[:protocol.lineno - 1]).rstrip("\r\n") + "\n\n\n"
                    + "".join(lines[protocol.end_lineno:]).lstrip("\r\n"))
        else:
            text = text.replace("from dataclasses import replace\n", "", 1)
        restored = text.encode("utf-8")

        reconstructed.append({
            "path": path, "repair5_sha256": expected[path],
            "qualification": "Original bytes reconstructed only by removing the four new test nodes, new Protocol and imports; exact repair5 SHA-256 verified.",
        })
        restored_path = SCRATCH / ("p5t4-generation-reconstructed-" + Path(path).name)
        restored_path.write_bytes(restored)

    expected = current

        elapsed = time.monotonic() - started

            raw_path.write_bytes(content)

for path in SOURCES + TESTS:
    raw = (ROOT / path).read_bytes()
    text = raw.decode("utf-8")
    tree = ast.parse(text)
    lines = text.splitlines()
    definitions = [n for n in ast.walk(tree)
                   if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef))]
    loops = [n for n in ast.walk(tree) if isinstance(n, (ast.For, ast.While, ast.AsyncFor))]
    assert len(lines) <= 500 and all(ast.get_docstring(n) for n in definitions), path
    assert all(lines[n.lineno - 2].lstrip().startswith("#") for n in loops), path
    structure.append({**reference(ROOT / path), "physical_lines": len(lines),
                      "documented_definitions": len(definitions), "commented_loops": len(loops)})
    if path not in SOURCES:
        continue
    entry = coverage["files"][path.replace("/", "\\")]
    summaries[path] = entry["summary"]
    statements = set(entry["executed_lines"]) | set(entry["missing_lines"])
    executed = set(entry["executed_lines"])
    if len(executed) / len(statements) < .9:
        failures.append(path + ": module statements below 90%")
    if entry["summary"]["percent_branches_covered"] < 75:
        failures.append(path + ": module branches below 75%")
    parents = {child: parent for parent in ast.walk(tree) for child in ast.iter_child_nodes(parent)}
    for node in definitions:
        ancestors = []
        parent = parents.get(node)
        while parent is not None:
            if isinstance(parent, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
                ancestors.append(parent.name)
            parent = parents.get(parent)
        name = ".".join([*reversed(ancestors), node.name])
        start = min([node.lineno, *(item.lineno for item in node.decorator_list)])
        selected = {line for line in statements if start <= line <= node.end_lineno}
        missing = selected - executed
        percent = 100 * (len(selected) - len(missing)) / len(selected) if selected else 100
        row = {"path": path, "name": name, "start": start, "end": node.end_lineno,
               "statements": len(selected), "missing": sorted(missing), "percent": percent}
        ranges.append(row)
        if percent < 90:
            failures.append(row)
