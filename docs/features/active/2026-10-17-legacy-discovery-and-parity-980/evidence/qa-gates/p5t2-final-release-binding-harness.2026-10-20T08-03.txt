def sha(data):
    """Hash exact source, script or evidence bytes."""
    return hashlib.sha256(data).hexdigest()


def read(path, expected=None):
    """Read an existing JSON receipt without locale or terminal-newline assumptions."""
    raw = path.read_bytes()
    if expected is not None:
        assert sha(raw) == expected, path
    return json.loads(raw), {"path": path.as_posix(), "sha256": sha(raw)}


def write(path, document):
    """Create a new LF UTF-8 release document without changing historical receipts."""
    assert not path.exists(), path
    path.write_text(json.dumps(document, indent=2, sort_keys=True) + "\n", encoding="utf-8", newline="\n")
