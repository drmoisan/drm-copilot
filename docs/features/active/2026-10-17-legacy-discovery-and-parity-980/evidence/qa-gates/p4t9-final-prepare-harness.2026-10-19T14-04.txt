now = datetime.datetime.now(datetime.timezone.utc)
stamp = now.strftime('%Y-%m-%dT%H-%M')


def load(relative):
    """Read one immutable feature evidence file."""
    return json.loads((evidence / relative).read_bytes())


def ref(path):
    """Bind exact working bytes without claiming Git-normalized equality."""
    path = Path(path)
    return {'path': path.as_posix(), 'sha256': hashlib.sha256(path.read_bytes()).hexdigest()}


def write(path, value):
    """Create a new LF-only readable JSON artifact."""
    assert not path.exists(), path
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + '\n', encoding='utf-8', newline='\n')

def routes(value):
    """Find the existing exact routed authority without writing checkpoint state."""
    if isinstance(value, dict):
        if value.get('delegation_id') == 'workspace_operations_p4t9_final_gates':
            yield value
        for item in value.values():
            yield from routes(item)
    elif isinstance(value, list):
        for item in value:
            yield from routes(item)
