checked = {}


def sha(data: bytes) -> str:
    """Return exact byte SHA-256 without normalization or I/O."""
    return hashlib.sha256(data).hexdigest()


def bound(reference: dict) -> bytes:
    """Read a frozen proof and retain its exact published hash binding."""
    data = Path(reference["path"]).read_bytes()
    assert sha(data) == reference["sha256"], reference["path"]
    checked[reference["path"]] = reference["sha256"]
    return data


def raw(reference: dict) -> bytes:
    """Verify one retained raw stream under the worker's established scratch root."""
    name = reference["path"].replace("\\", "/")
    assert name.startswith("SCRATCH/")
    data = (SCRATCH / name.removeprefix("SCRATCH/")).read_bytes()
    assert sha(data) == reference["sha256"], name
    return data
