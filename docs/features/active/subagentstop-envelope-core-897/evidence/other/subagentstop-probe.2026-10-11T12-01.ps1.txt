param(
    [int]$ExitCode = 0,
    [string]$LogDir,
    [string]$Mode = 'exit0'
)
$ErrorActionPreference = 'Stop'
$stdinText = ''
try { $stdinText = [Console]::In.ReadToEnd() } catch { $stdinText = "<stdin read error: $($_.Exception.Message)>" }

$envHookInput = [Environment]::GetEnvironmentVariable('CLAUDE_HOOK_INPUT')
$claudeEnvNames = @(Get-ChildItem Env: | Where-Object { $_.Name -like 'CLAUDE*' } | ForEach-Object { $_.Name } | Sort-Object)

$parsedKeys = @()
$lastMsg = $null
$hasOutput = $false
$stopHookActive = $null
try {
    $obj = $stdinText | ConvertFrom-Json -ErrorAction Stop
    $parsedKeys = @($obj.PSObject.Properties.Name)
    $lastMsg = $obj.last_assistant_message
    $hasOutput = $obj.PSObject.Properties.Name -contains 'output'
    $stopHookActive = $obj.stop_hook_active
} catch { $parsedKeys = @("<parse error: $($_.Exception.Message)>") }

$stamp = [DateTime]::UtcNow.ToString('yyyyMMddTHHmmssfff')
$record = [ordered]@{
    mode                       = $Mode
    utc                        = $stamp
    stdin_length               = $stdinText.Length
    stdin_raw                  = $stdinText
    stdin_top_level_keys       = $parsedKeys
    stdin_has_output_key       = $hasOutput
    last_assistant_message     = $lastMsg
    stop_hook_active           = $stopHookActive
    env_CLAUDE_HOOK_INPUT_set  = [bool]($null -ne $envHookInput)
    env_CLAUDE_HOOK_INPUT_len  = if ($null -ne $envHookInput) { $envHookInput.Length } else { 0 }
    env_CLAUDE_names           = $claudeEnvNames
    planned_exit               = $ExitCode
}
$record | ConvertTo-Json -Depth 6 | Set-Content -LiteralPath (Join-Path $LogDir "probe-$Mode-$stamp.json") -Encoding utf8

if ($ExitCode -ne 0) {
    $marker = Join-Path $LogDir "blocked-once-$Mode.marker"
    if (-not (Test-Path -LiteralPath $marker)) {
        Set-Content -LiteralPath $marker -Value $stamp
        [Console]::Error.WriteLine("PROBE_BLOCK_$Mode`: before stopping, reply with the single token BLOCKED-ACK-$Mode.")
        exit $ExitCode
    }
}
exit 0
