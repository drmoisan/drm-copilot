def sha(raw):
    """Hash exact stored bytes without textual normalization."""
    return hashlib.sha256(raw).hexdigest()


def read(path, digest):
    """Read an immutable owner receipt after checking its announced hash."""
    raw = path.read_bytes()
    assert sha(raw) == digest, path
    return json.loads(raw), {"path": path.as_posix(), "sha256": digest}


def save(path, value):
    """Create a portable LF UTF-8 authored receipt without overwriting evidence."""
    assert not path.exists(), path
    text = json.dumps(value, indent=2, sort_keys=True) + "\n"
    assert not re.search(r"[A-Za-z]:[\\/]", text), path
    path.write_text(text, encoding="utf-8", newline="\n")
    return {"path": path.as_posix(), "sha256": sha(path.read_bytes())}


def extract_coverage(archive_ref, identifier):
    """Decode already retained raw coverage; preserve its byte identity separately."""
    archive, _ = read(Path(archive_ref["path"]), archive_ref["sha256"])
    entries = [entry for entry in archive["entries"] if entry["identifier"] == identifier]
    assert len(entries) == 1
    entry = entries[0]
    assert entry["encoding"] == "gzip+base64"
    raw = gzip.decompress(base64.b64decode(entry["data"]))
    assert sha(raw) == entry["sha256"] and len(raw) == entry["bytes"]
    return json.loads(raw), {"archive": archive_ref, "identifier": identifier,
                            "raw_sha256": sha(raw), "raw_bytes": len(raw)}
