def digest(data):
    return sha256(data).hexdigest()


def ref(path):
    return {"path": Path(path).as_posix(), "sha256": digest(Path(path).read_bytes())}


def load(path):
    return json.loads(Path(path).read_bytes())


def raw_path(portable_path):
    assert portable_path.startswith("SCRATCH/")
    path = (WORKER / portable_path.removeprefix("SCRATCH/")).resolve()
    assert path.is_relative_to(WORKER)
    return path


def portable(text):
    for path, name in ((ROOT, "WORKSPACE"), (WORKER, "SCRATCH")):
        text = re.sub(re.escape(str(path)), lambda _: name, text, flags=re.IGNORECASE)
        text = re.sub(re.escape(path.as_posix()), lambda _: name, text, flags=re.IGNORECASE)
    return text


def verify_refs(value):
    if isinstance(value, dict):
        if "path" in value and "sha256" in value and value["path"].startswith("docs/"):
            assert ref(value["path"])["sha256"] == value["sha256"], value["path"]
        for item in value.values():
            verify_refs(item)
    elif isinstance(value, list):
        for item in value:
            verify_refs(item)

assert ref(HANDOFF.with_suffix(".md"))["sha256"] == "2c2b6b5e813fc946d9db4bd906c874703e2f0055934be0e250fe322f9348ccae"

assert ordered["candidate_manifest_sha256"] == handoff["candidate"]["sha256"]
assert digest(raw_path(ordered["candidate_manifest"]).read_bytes()) == ordered["candidate_manifest_sha256"]

assert expected == {item["path"]: item["sha256"] for item in candidate["files"]}

source_rows = []
for item in handoff["source_files"]:
    path = item["path"]
    working = Path(path).read_bytes()
    index = subprocess.check_output(["git", "show", ":" + path])
    committed = subprocess.check_output(["git", "show", commit + ":" + path])
    assert working == index == committed
    assert digest(working) == item["sha256"] == item["index_sha256"] == item["commit_sha256"]
    assert b"\r" not in working and len(working.splitlines()) <= 500

snapshots = load(handoff["executed_harnesses"]["path"])["snapshots"]
harness_rows = []
for item in snapshots:
    raw = base64.b64decode(item["executed_content_base64"], validate=True)
    assert digest(raw) == item["executed_sha256"]
    assert raw == raw_path(item["executed_path"]).read_bytes()
    companion = Path(item["readable_lf_companion"]["path"]).read_bytes()
    assert raw.replace(b"\r\n", b"\n") == companion
    assert digest(companion) == item["readable_lf_companion"]["sha256"]
    if item["role"] == "final-gate-harness":
        assert ordered["harness_hashes"][item["executed_path"]] == digest(raw)
    harness_rows.append({"path": item["executed_path"], "role": item["role"],
                         "raw_sha256": digest(raw), "lf_sha256": digest(companion),
                         "raw_crlf_count": raw.count(b"\r\n"), "exact_base64_and_scratch_match": True})
assert len(harness_rows) == 4 and len(ordered["harness_hashes"]) == 3
