"""Bind completed P4-T7 proof without re-running gates or granting task acceptance."""
import ast
import base64
import datetime
import hashlib
import json
from pathlib import Path
import re
import subprocess
import sys

root = Path.cwd()
scratch = Path(__file__).resolve().parent
qa = root / 'docs/features/active/2026-10-17-legacy-discovery-and-parity-980/evidence/qa-gates'
stamp = datetime.datetime.now(datetime.timezone.utc).strftime('%Y-%m-%dT%H-%M')
ordered_path = root / sys.argv[1]
ordered = json.loads(ordered_path.read_bytes())
candidate_path = scratch / ordered['candidate_manifest'].removeprefix('SCRATCH/')
candidate = json.loads(candidate_path.read_bytes())


def sha(content):
    return hashlib.sha256(content).hexdigest()


def ref(path):
    return {'path': path.relative_to(root).as_posix(), 'sha256': sha(path.read_bytes())}


def write(path, value):
    assert not path.exists(), path
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + '\n', encoding='utf-8', newline='\n')


assert sha(candidate_path.read_bytes()) == ordered['candidate_manifest_sha256']
assert [(item['stage'], item['label']) for item in ordered['receipts']] == [(1, 'format'), (2, 'lint'), (3, 'types'), (4, 'architecture'),
    (5, 'tests-shared-reverse'), (5, 'tests-node-reverse'), (5, 'tests-python-reverse'), (5, 'tests-dotnet-reverse'), (5, 'tests-integration-reverse'), (6, 'contracts'), (7, 'native')]
assert len(ordered['test_nodes']) == len(set(ordered['test_nodes'])) == sum(candidate['expected_test_counts'].values())
receipts = []
previous_finished = None
for entry in ordered['receipts']:
    path = root / entry['path']
    assert sha(path.read_bytes()) == entry['sha256']
    receipt = json.loads(path.read_bytes())
    assert receipt['exit_code'] == 0 and receipt['source_before'] == receipt['source_after'] == ordered['source_hashes']
    assert receipt['harness_hashes'] == ordered['harness_hashes']
    if previous_finished is not None:
        assert previous_finished <= receipt['started_at']
    previous_finished = receipt['finished_at']
    for stream in ('raw_stdout', 'raw_stderr'):
        raw = receipt[stream]
        assert sha((scratch / raw['path'].removeprefix('SCRATCH/')).read_bytes()) == raw['sha256']
    receipts.append({**entry, 'seconds': receipt['seconds'], 'markdown': ref(path.with_suffix('.md'))})
contracts = json.loads((root / ordered['receipts'][-2]['path']).read_bytes())
contract_count = int(re.findall(r'(\d+) passed', contracts['stdout'])[-1])
native = json.loads((root / ordered['native_path']).read_bytes())
assert len(native['checks']) == 16 and native['source_tree_unchanged'] and native['git_status_unchanged']
assert native['publication'] == 'refused-unregistered-adapter'
assert sha((scratch / ordered['raw_native']['path'].removeprefix('SCRATCH/')).read_bytes()) == ordered['raw_native']['sha256']
head = subprocess.run(['git', 'rev-parse', 'HEAD'], capture_output=True, check=True).stdout.decode().strip()
files, unit_ranges, coverage_refs = [], [], []
module_coverage = {}
for batch in ordered['coverage_batches']:
    path = root / batch['coverage_path']
    assert sha(path.read_bytes()) == batch['coverage_sha256']
    raw_coverage_bytes = (scratch / batch['raw_coverage']['path'].removeprefix('SCRATCH/')).read_bytes()
    assert sha(raw_coverage_bytes) == batch['raw_coverage']['sha256']
    raw_coverage_path = qa / ('p4t7-final-' + batch['batch'] + '-original-coverage.' + stamp + '.json')
    assert not raw_coverage_path.exists()
    raw_coverage_path.write_bytes(raw_coverage_bytes)
    coverage = json.loads(path.read_bytes())
    coverage_refs.append({'batch': batch['batch'], **ref(path), 'totals': coverage['totals'], 'raw': batch['raw_coverage'],
                          'test_count': len(batch['test_nodes']), 'test_module_order': batch['test_module_order'], 'canonical_original': ref(raw_coverage_path)})
    for relative, values in coverage['files'].items():
        assert relative not in module_coverage
        module_coverage[relative] = values['summary']
        tree = ast.parse((root / relative).read_bytes())
        def collect(nodes, prefix=''):
            for node in nodes:
                if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
                    family = 'classes' if isinstance(node, ast.ClassDef) else 'functions'
                    qualified = prefix + node.name
                    measured = values[family][qualified]['summary']
                    assert not measured['num_statements'] or measured['percent_statements_covered'] >= 90
                    unit_ranges.append({'path': relative, 'name': qualified, 'kind': type(node).__name__,
                        'start': node.lineno, 'end': node.end_lineno, 'summary': measured})
                    collect(node.body, qualified + '.')
        collect(tree.body)
assert len(module_coverage) == 6
for entry in candidate['files']:
    relative = entry['path']
    content = (root / relative).read_bytes()
    assert sha(content) == entry['sha256'] == ordered['source_hashes'][relative]
    index = subprocess.run(['git', 'show', ':' + relative], capture_output=True, check=True).stdout
    committed = subprocess.run(['git', 'show', 'HEAD:' + relative], capture_output=True, check=True).stdout
    assert content.replace(b'\r\n', b'\n') == index == committed
    blob = subprocess.run(['git', 'hash-object', '--no-filters', '--stdin'], input=content, capture_output=True, check=True).stdout.decode().strip()
    filtered = subprocess.run(['git', 'hash-object', '--path=' + relative, '--stdin'], input=content, capture_output=True, check=True).stdout.decode().strip()
    attributes = subprocess.run(['git', 'check-attr', 'text', 'eol', '--', relative], capture_output=True, check=True).stdout.decode().strip()
    files.append({'path': relative, 'sha256': sha(content), 'lines': len(content.splitlines()), 'working_git_blob': blob,
        'filtered_git_blob': filtered, 'index_sha256': sha(index), 'commit_sha256': sha(committed), 'commit': head,
        'git_attributes': attributes, 'stream_relation': 'identical' if content == index else 'CRLF-to-LF-only normalization'})
baseline = root / candidate['baseline']['path']
assert sha(baseline.read_bytes()) == candidate['baseline']['sha256']
preserved = json.loads(baseline.read_bytes())['preserved_files']
for entry in preserved:
    assert sha((root / entry['path']).read_bytes()) == entry['sha256']
snapshot_inputs = dict(ordered['harness_hashes'])
snapshot_inputs['SCRATCH/' + Path(__file__).name] = sha(Path(__file__).read_bytes())
for name in ('issue-980-p4t7-final-adapt.py', 'issue-980-p4t7-final-prepare.py', 'issue-980-p4t7-final-preparation-correction.py', 'issue-980-p4t7-final-prepare2.py', 'issue-980-p4t7-final-preparation-delta.py', 'issue-980-p4t7-final-native.prepared-original.py', 'issue-980-p4t7-final-handoff-prepare.py'):
    snapshot_inputs['SCRATCH/' + name] = sha((scratch / name).read_bytes())
harnesses = []
for relative, digest in snapshot_inputs.items():
    source = scratch / relative.removeprefix('SCRATCH/')
    content = source.read_bytes()
    assert sha(content) == digest
    companion = qa / (source.stem.removeprefix('issue-980-') + '-harness.' + stamp + '.txt')
    assert not companion.exists()
    companion.write_bytes(content.replace(b'\r\n', b'\n'))
    harnesses.append({'executed_path': relative, 'executed_sha256': digest,
        'role': 'final-gate-harness' if relative in ordered['harness_hashes'] else ('unexecuted-preserved-preparation' if 'prepared-original' in relative else 'preparation-or-handoff-binding'),
        'encoding': 'base64', 'executed_content_base64': base64.b64encode(content).decode('ascii'),
        'readable_lf_companion': ref(companion), 'normalization': 'Base64 preserves exact executed bytes; readable companion changes CRLF to LF only.'})
harness_path = qa / ('p4t7-final-executed-harnesses.' + stamp + '.json')
write(harness_path, {'task': 'P4-T7', 'timestamp': stamp, 'snapshots': harnesses})
candidate_copy = qa / ('p4t7-final-candidate-binding.' + stamp + '.json')
assert not candidate_copy.exists()
candidate_copy.write_bytes(candidate_path.read_bytes())
range_path = qa / ('p4t7-final-coverage-ranges.' + stamp + '.json')
write(range_path, {'task': 'P4-T7', 'timestamp': stamp, 'module_summaries': module_coverage, 'unit_ranges': unit_ranges,
    'coverage_batches': coverage_refs, 'baseline_disposition': 'All six modules are new at the accepted P4-T6 baseline. Each module has 100% measured statements/branches and every class/callable meets its required floor. Original shared repair and format-parser proofs remain immutable; all predecessor streams are unchanged.'})
for entry in candidate['subset_manifests'] + [candidate['review']] + ([candidate['confirmation']] if candidate.get('confirmation') else []):
    assert sha((root / entry['path']).read_bytes()) == entry['sha256']
handoff = qa / ('p4t7-final-handoff.' + stamp + '.json')
document = {'task': 'P4-T7', 'timestamp': stamp, 'state': 'EXECUTOR_ACCEPTANCE_PENDING', 'checked_candidate_commit': head,
    'ordered_stages': ref(ordered_path), 'ordered_receipts': receipts, 'candidate': ref(candidate_copy), 'source_files': files,
    'subset_manifests': candidate['subset_manifests'], 'retained_review': candidate['review'], 'bounded_confirmation': candidate.get('confirmation'),
    'executed_harnesses': ref(harness_path), 'coverage_ranges': ref(range_path),
    'test_cases': {**candidate['expected_test_counts'], 'total': len(ordered['test_nodes']), 'contracts': contract_count, 'native_assertions': 16},
    'native_result': ref(root / ordered['native_path']), 'shared_baseline': ref(baseline), 'preserved_predecessors': len(preserved),
    'toolchain_deltas': {'new_lint_findings': 0, 'new_type_diagnostics': 0, 'new_test_failures': 0},
    'limitations': ['Finite package.json, npm lock/shrinkwrap v2/v3, requirements/setup.cfg, MSBuild declarations, packages.config and NuGet lock v1 syntax only.',
        'Installed state, effective resolution, graph binding, conditions and execution remain unverified or unresolved.',
        'Unsafe locations, versions, conditions and marker values are redacted; metadata never supplies an evaluated graph.',
        'Dispatch and successful publication remain P4-T9; the native proof verifies expected closed-registry refusal.',
        'Native Windows Python/Git only; no installed coding-agent host, non-Windows, package installation/restore/execution proof.',
        'Independent task/AC acceptance remains executor-owned.'],
    'changed_line_coverage': 'All six newly added production modules have 100% measured statements/branches; changed responsibilities retain or improve their frozen focused baseline. No predecessor file changed.',
    'overall_repository_coverage': 'Not remeasured for this atomic task; the feature-wide gate remains outstanding. This receipt claims the complete six-module task population only.'}

write(handoff, document)
table = '\n'.join('| ' + Path(path).name + ' | ' + str(item['covered_lines']) + '/' + str(item['num_statements']) + ' | ' +
    str(item['covered_branches']) + '/' + str(item['num_branches']) + ' |' for path, item in module_coverage.items())

Seven ordered stages passed on 14 bound source/test files.\n\nScoped Black/Ruff, whole-repository Pyright, architecture/size/purity/isolation, five isolated reversed module-order processes ({len(ordered["test_nodes"])} cases), {contract_count} applicable contracts, and 16 native Windows/Git assertions passed. Exact separate raw streams, real elapsed times, full node IDs, source/Git bindings and executed harnesses are retained.\n\n| Module | Statements | Branches |\n|---|---:|---:|\n{table}\n\nThe JSON records actual coverage for every class/callable, including uncovered guards. No installation, restore, dependency execution, evaluated graph, behavioral coverage, installed-host or non-Windows claim is made. Independent acceptance remains pending.\n'
handoff.with_suffix('.md').write_text(markdown, encoding='utf-8', newline='\n')
print(json.dumps({'handoff': ref(handoff), 'markdown': ref(handoff.with_suffix('.md')), 'case_counts': document['test_cases'],
                  'unit_ranges': len(unit_ranges), 'source_files': len(files), 'modules': len(module_coverage)}), flush=True)
