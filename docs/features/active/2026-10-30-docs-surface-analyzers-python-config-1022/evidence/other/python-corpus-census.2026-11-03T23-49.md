## Helper text (SCRATCH/issue-1022-p2-census.ps1)

```powershell
#Requires -Version 7.0
[CmdletBinding()]
param(
    [Parameter(Mandatory = $true)] [string] $WorktreeRoot,
    [Parameter(Mandatory = $true)] [string] $Corpus,
    [Parameter(Mandatory = $true)] [string[]] $Code,
    [string[]] $InTest = @(),
    [string[]] $Declaration = @()
)
Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'
$Code = @($Code | ForEach-Object { $_ -split ',' } | Where-Object { $_ })
$InTest = @($InTest | ForEach-Object { $_ -split ',' } | Where-Object { $_ })
Import-Module -Name (Join-Path -Path $WorktreeRoot -ChildPath '.claude/lib/docs/DocsJson.psm1') -Force -ErrorAction Stop
$path = Join-Path -Path $WorktreeRoot -ChildPath $Corpus
$bytes = [System.IO.File]::ReadAllBytes($path)
$crCount = @($bytes | Where-Object { $_ -eq 13 }).Count
$nonAscii = @($bytes | Where-Object { $_ -gt 127 }).Count
Write-Output "FILE $Corpus bytes=$($bytes.Length) cr_bytes=$crCount non_ascii_bytes=$nonAscii"
$text = [System.Text.Encoding]::UTF8.GetString($bytes)
$parsed = ConvertFrom-DocsJsonText -Text $text
Write-Output "PARSE findings=$($parsed.Findings.Count)"
foreach ($finding in $parsed.Findings) { Write-Output "  $($finding['code']) $($finding['pointer']) $($finding['message'])" }
if ($null -eq $parsed.Value) { exit 1 }
$value = $parsed.Value
$elements = @($value['expected_elements'])
$diagnostics = @($value['expected_diagnostics'])
Write-Output "COUNTS inputs=$(@($value['inputs'].Keys).Count) expected_elements=$($elements.Count) expected_diagnostics=$($diagnostics.Count)"
$total = 0
$missing = 0
foreach ($entry in $Code) {
    $count = @($diagnostics | Where-Object { $_['code'] -ceq $entry }).Count
    $total++
    if ($count -lt 1) { $missing++ }
    Write-Output "ROW $entry $count"
}
foreach ($entry in $InTest) {
    $count = @($diagnostics | Where-Object { $_['code'] -ceq $entry }).Count
    Write-Output "IN-TEST $entry corpus_entries=$count"
}
$known = [System.Collections.Generic.HashSet[string]]::new([string[]] ($Code + $InTest), [System.StringComparer]::Ordinal)
$extra = @($diagnostics | ForEach-Object { $_['code'] } | Where-Object { -not $known.Contains($_) } | Sort-Object -Unique)
Write-Output "OTHER-CODES $(if ($extra.Count -eq 0) { 'none' } else { $extra -join ',' })"
$grammar = @($diagnostics | Where-Object { $_['code'] -cnotmatch '^[A-Z][A-Z0-9_]*$' }).Count
Write-Output "GRAMMAR codes_outside_1020_grammar=$grammar"
foreach ($wanted in $Declaration) {
    $hit = @($elements | Where-Object { $_['declaration'] -ceq $wanted }).Count
    Write-Output "DECLARATION [$wanted] count=$hit"
}
Write-Output "SUMMARY rows=$total rows_below_one=$missing"
```
