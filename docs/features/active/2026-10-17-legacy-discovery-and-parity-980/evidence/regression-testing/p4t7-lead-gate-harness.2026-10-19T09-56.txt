"""Preserve exact ordered shared-contract command streams and immutable source bindings."""
import datetime
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys
import time

repo, scratch = Path.cwd(), Path(__file__).parent
evidence = repo / "docs/features/active/2026-10-17-legacy-discovery-and-parity-980/evidence/regression-testing"
stage, label, *command = sys.argv[1:]
paths = ["packages/discovery-runtime/src/drm_copilot_discovery/analyzer/dependency_inputs.py",
         "tests/discovery_runtime/analyzer/test_dependency_inputs.py"]


def bindings():
    return {path: {"sha256": hashlib.sha256((repo / path).read_bytes()).hexdigest(),
                   "physical_lines": len((repo / path).read_text("utf-8").splitlines())} for path in paths}


def sanitize(value):
    for path, replacement in ((scratch, "SCRATCH"), (repo, ".")):
        for spelling in (str(path), path.as_posix()):
            value = re.sub(re.escape(spelling), lambda match: replacement, value, flags=re.I)
    return value


before = bindings()
started = datetime.datetime.now(datetime.timezone.utc)
timer = time.monotonic()
environment = dict(os.environ, PYTHONIOENCODING="utf-8", COVERAGE_FILE=str(scratch / "issue-980-p4t7-lead-coverage.sqlite"))
result = subprocess.run(command, cwd=repo, env=environment, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
duration = round(time.monotonic() - timer, 3)
finished = datetime.datetime.now(datetime.timezone.utc)
stamp = finished.strftime("%Y-%m-%dT%H-%M")
streams = {}
for kind, raw in (("stdout", result.stdout), ("stderr", result.stderr)):
    filename = "issue-980-" + label + "-" + finished.strftime("%H%M%S") + "." + kind + ".bin"
    (scratch / filename).write_bytes(raw)
    streams[kind] = {"scratch_identifier": "SCRATCH/" + filename, "raw_sha256": hashlib.sha256(raw).hexdigest(),
                     "raw_bytes": len(raw), "presentation": sanitize(raw.decode("utf-8", errors="replace"))}
record = {"stage": stage, "label": label, "started": started.isoformat(), "finished": finished.isoformat(),
          "command": sanitize(subprocess.list2cmdline(command)), "exit_code": result.returncode,
          "duration_seconds": duration, "source_before": before, "source_after": bindings(), "streams": streams,
          "accepted_baseline": "28b03c185f32fdeb9ccfdafd1bb7decce959161d", "scope": "P4-T7 shared dependency contract only"}
prefix = evidence / (label + "." + stamp)
json_path, md_path = Path(str(prefix) + ".json"), Path(str(prefix) + ".md")
assert not json_path.exists() and not md_path.exists()
json_path.write_bytes((json.dumps(record, indent=2) + "\n").encode())
output = "\n".join(streams[kind]["presentation"] for kind in ("stdout", "stderr"))
rendered = "\n".join(line.rstrip() for line in output.splitlines()).rstrip()
md_path.write_bytes((f"# {label}\n\nTimestamp: {stamp}\nCommand: {record['command']}\nEXIT_CODE: {result.returncode}\n"
                    f"Output Summary: scoped {stage}; duration {duration} seconds. Raw stdout/stderr hashes and SCRATCH-relative locations are separate from sanitized presentation.\n\n```text\n{rendered}\n```\n").encode())
sys.stdout.reconfigure(encoding="utf-8")
print(output)
print(json_path.relative_to(repo).as_posix())
sys.exit(result.returncode)
