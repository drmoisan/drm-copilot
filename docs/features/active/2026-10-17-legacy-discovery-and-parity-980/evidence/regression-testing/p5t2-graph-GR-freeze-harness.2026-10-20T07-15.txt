def sha(raw: bytes) -> str:
    """Return an exact byte identity without interpreting content."""
    return hashlib.sha256(raw).hexdigest()


def ref(path: Path) -> dict[str, object]:
    """Bind an existing workspace artifact using a relative location."""
    raw = path.read_bytes()
    return {"path": path.relative_to(ROOT).as_posix(),
            "sha256": sha(raw), "bytes": len(raw)}


def save(name: str, value: object) -> Path:
    """Write a new LF-only canonical JSON artifact at the current write minute."""
    path = OUT / f"{name}.{STAMP}.json"
    path.write_bytes((json.dumps(value, indent=2, sort_keys=True) + "\n").encode())
    return path


def definitions(tree: ast.AST, prefix: str = "") -> dict[str, ast.AST]:
    """Index qualified definitions for raw-range coverage and exact comparisons."""
    result: dict[str, ast.AST] = {}
    for node in ast.iter_child_nodes(tree):
        if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
            name = prefix + node.name
            result[name] = node
            result.update(definitions(node, name + "."))
    return result

# Reuse the immutable pre-repair source archive to prove the exact authorized delta.
old_archive = json.loads((ROOT / PRE["raw_archive"]["path"]).read_text(encoding="utf-8"))
old_sources = {}
for path in SOURCES + TESTS[:-1]:
    member = old_archive["members"][path]
    raw = gzip.decompress(base64.b64decode(member["data"]))
    assert sha(raw) == member["sha256"]
    old_sources[path] = raw.decode()
for path in SOURCES:
    before = definitions(ast.parse(old_sources[path]))
    after = definitions(ast.parse((ROOT / path).read_text(encoding="utf-8")))
    for name, old in before.items():
        if path.endswith("/parity_metrics.py") and name in {
                "ParityProjection", "ParityProjection.axes"}:
            continue
        assert ast.dump(old) == ast.dump(after[name]), (path, name)
assertions_retained = {}
for path in TESTS[:-1]:
    old = ast.parse(old_sources[path])
    new = ast.parse((ROOT / path).read_text(encoding="utf-8"))
    old_asserts = [ast.dump(node) for node in ast.walk(old)
                   if isinstance(node, ast.Assert)]
    new_asserts = [ast.dump(node) for node in ast.walk(new)
                   if isinstance(node, ast.Assert)]
    assert old_asserts == new_asserts, path
    assertions_retained[path] = len(old_asserts)
old_helper = definitions(ast.parse(old_sources[TESTS[0]]))["_assessed_projection"]
new_helper = definitions(ast.parse((ROOT / TESTS[0]).read_text(encoding="utf-8")))["assessed_projection"]
old_helper.name = new_helper.name
assert ast.dump(old_helper) == ast.dump(new_helper)

# Current execution and retained unchanged-body proof remain separately inspectable.
qualified = {}
retained = {item["source"]: item for item in PRE["retained_unchanged_body_proof"]}
for path in SOURCES:
    raw = next(value for name, value in coverage["files"].items()
               if name.replace("\\", "/") == path)
    executed, missing = set(raw["executed_lines"]), set(raw["missing_lines"])
    retained_lines = set()
    if path in retained:
        assert sha((ROOT / path).read_bytes()) == next(
            row["sha256"] for row in PRE["sources_and_tests"] if row["path"] == path)
        for item in retained[path]["exact_ast_unchanged"]:
            retained_lines.update(item["mapped_covered_lines"])
    measured = executed | missing
    supported = executed | retained_lines
    ranges = []
    for name, node in definitions(ast.parse((ROOT / path).read_text(encoding="utf-8"))).items():
        statements = {line for line in measured
                      if node.lineno <= line <= node.end_lineno}
        absent = statements - supported
        percentage = 100.0 if not statements else 100 * (1 - len(absent) / len(statements))
        assert percentage >= 90, (path, name, percentage, sorted(absent))
        ranges.append({"name": name, "start": node.lineno, "end": node.end_lineno,
                       "statements": len(statements), "current_covered": len(statements & executed),
                       "retained_additional": sorted(statements & (retained_lines - executed)),
                       "qualified_line_percentage": percentage, "unsupported_lines": sorted(absent)})
    try:
        base_text = subprocess.check_output(
            ["git", "show", f"2b069afb0d88ae1a3c17ef10f7a52c3936800462:{path}"],
            cwd=ROOT, stderr=subprocess.DEVNULL).decode()
    except subprocess.CalledProcessError:
        base_text = ""
    current_lines = (ROOT / path).read_text(encoding="utf-8").splitlines()
    unchanged = set()
    for block in difflib.SequenceMatcher(
            a=base_text.splitlines(), b=current_lines, autojunk=False).get_matching_blocks():
        unchanged.update(range(block.b + 1, block.b + block.size + 1))
    changed = measured - unchanged
    assert changed <= executed, (path, sorted(changed - executed))
    qualified[path] = {"raw_summary": raw["summary"], "ranges": ranges,
                       "changed_executable_lines": sorted(changed),
                       "changed_lines_currently_covered": True,
                       "retained_proof": "See pre-repair freeze exact-AST body mappings",
                       "unsupported_module_lines": sorted(measured - supported)}
    assert not qualified[path]["unsupported_module_lines"], path
