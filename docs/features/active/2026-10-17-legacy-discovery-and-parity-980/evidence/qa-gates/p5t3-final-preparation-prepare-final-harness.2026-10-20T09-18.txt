def sha(raw):
    """Return the exact byte identity of a preserved stream."""
    return hashlib.sha256(raw).hexdigest()


def bound(path, expected):
    """Read one immutable receipt only after its exact digest matches."""
    raw = path.read_bytes()
    assert sha(raw) == expected, path
    return json.loads(raw), {"path": path.as_posix(), "sha256": expected}


def save(path, value):
    """Create a new authored UTF-8 LF receipt without replacing prior evidence."""
    assert not path.exists(), path
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n",
                    encoding="utf-8", newline="\n")


def inert_hygiene(raw, name):
    """Check only script syntax, explicit text decoding, size and readable bytes."""
    tree = ast.parse(raw)
    assert len(raw.splitlines()) <= 500, name
    assert b"\r" not in raw and raw.endswith(b"\n") and not raw.endswith(b"\n\n"), name
    assert not re.search(rb"[ \t]+\r?$", raw, re.M), name
    # Binary readers preserve original bytes; text readers must declare their encoding.
    for node in ast.walk(tree):
        if not isinstance(node, ast.Call):
            continue
        function = ast.unparse(node.func)
        keywords = {item.arg: item.value for item in node.keywords}
        if function.endswith((".read_text", ".write_text")):
            assert "encoding" in keywords, (name, function, node.lineno)
        if function.endswith(".open") or function == "open":
            mode = keywords.get("mode", node.args[0] if function.endswith(".open") and node.args else None)
            if function == "open":
                mode = keywords.get("mode", node.args[1] if len(node.args) > 1 else None)
            binary = isinstance(mode, ast.Constant) and isinstance(mode.value, str) and "b" in mode.value
            assert binary or "encoding" in keywords, (name, function, node.lineno)
    return {"physical_lines": len(raw.splitlines()), "syntax_only": True,
            "explicit_text_encoding_or_binary_reads": True, "lf_utf8": True}
