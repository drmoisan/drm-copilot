sha = lambda data: hashlib.sha256(data).hexdigest()


def checked(path, expected=None):
    raw = Path(path).read_bytes()
    if expected is not None:
        assert sha(raw) == expected, str(path)
    return json.loads(raw)


def ref(path):
    return {'path': Path(path).as_posix(), 'sha256': sha(Path(path).read_bytes())}


def resolve(record):
    return checked(record['path'], record['sha256'])


def source_hashes(mapping):
    return {path: value if isinstance(value, str) else value['sha256'] for path, value in mapping.items()}

def publish(stem, suffix, payload):
    now = dt.datetime.now(dt.timezone.utc)
    stamp = now.strftime('%Y-%m-%dT%H-%M')
    path = QA / (stem + '.' + stamp + suffix)
    assert not path.exists(), str(path)
    if isinstance(payload, dict):
        payload = {**payload, 'timestamp': stamp, 'published_utc': now.isoformat()}
        data = (json.dumps(payload, indent=2) + '\n').encode()
    else:
        data = payload
    assert b'\r' not in data and data.endswith(b'\n') and not data.endswith(b'\n\n')
    assert all(line.rstrip() == line for line in data.splitlines())
    path.write_bytes(data)
    return path
