"""Reconcile immutable pure/planning proofs without executing another suite."""
from __future__ import annotations

import ast
import base64
import gzip
import hashlib
import json
import re
import sys
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path.cwd()
SCRATCH = Path(__file__).parent
FEATURE = Path("docs/features/active/2026-10-17-legacy-discovery-and-parity-980")
EVIDENCE = ROOT / FEATURE / "evidence"
OUT = EVIDENCE / "regression-testing"
STAMP = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H-%M")
ATTEMPT = sys.argv[1]
SOURCE = "packages/discovery-runtime/src/drm_copilot_discovery/completion_report.py"
TESTS = [
    "tests/discovery_runtime/test_completion_report.py",
    "tests/discovery_runtime/test_completion_planning.py",
]


def digest(raw: bytes) -> str:
    """Return exact source/evidence identity."""
    return hashlib.sha256(raw).hexdigest()


def reference(path: Path) -> dict[str, object]:
    """Describe an existing canonical stream without changing it."""
    raw = path.read_bytes()
    return {"path": path.relative_to(ROOT).as_posix(), "sha256": digest(raw), "bytes": len(raw)}


def read_json(path: Path) -> dict:
    """Decode retained UTF-8 JSON explicitly."""
    return json.loads(path.read_text(encoding="utf-8"))


def write_json(path: Path, value: object) -> None:
    """Write one authored LF-terminated readable artifact."""
    path.write_bytes((json.dumps(value, indent=2, sort_keys=True) + "\n").encode("utf-8"))


proofs = []
for prefix in ("p5t3-producer-pure2", "p5t3-producer-" + ATTEMPT):
    matches = sorted(OUT.glob(prefix + "-tests.*.json"))
    assert len(matches) == 1, matches
    path = matches[0]
    receipt = read_json(path)
    assert receipt["clean"] and receipt["exit_code"] == 0
    assert receipt["before"][SOURCE] == receipt["after"][SOURCE] == digest((ROOT / SOURCE).read_bytes())
    nodes = re.findall(r"^(tests/.+?) PASSED(?:\s|$)", receipt["stdout"], flags=re.M)
    assert nodes and len(nodes) == len(set(nodes))
    coverage = read_json(SCRATCH / (prefix + "-coverage.json"))
    entries = coverage["files"]
    assert len(entries) == 1
    proofs.append({"receipt": reference(path), "nodes": nodes, "coverage": next(iter(entries.values()))})
assert len(proofs[0]["nodes"]) == 35 and len(proofs[1]["nodes"]) == 19
assert not set(proofs[0]["nodes"]) & set(proofs[1]["nodes"])
statements = set(proofs[0]["coverage"]["executed_lines"]) | set(proofs[0]["coverage"]["missing_lines"])
executed = set()
branches = set()
covered_branches = set()
for proof in proofs:
    cov = proof["coverage"]
    assert statements == set(cov["executed_lines"]) | set(cov["missing_lines"])
    executed.update(cov["executed_lines"])
    branches.update(tuple(pair) for pair in cov["executed_branches"] + cov["missing_branches"])
    covered_branches.update(tuple(pair) for pair in cov["executed_branches"])
assert len(executed) / len(statements) >= .9
assert len(covered_branches) / len(branches) >= .75
source_tree = ast.parse((ROOT / SOURCE).read_text(encoding="utf-8"))
ranges = []
for node in source_tree.body:
    if not isinstance(node, (ast.FunctionDef, ast.ClassDef)):
        continue
    start = min([node.lineno, *(item.lineno for item in node.decorator_list)])
    selected = {line for line in statements if start <= line <= node.end_lineno}
    covered = selected & executed
    assert selected and len(covered) / len(selected) >= .9, node.name
    ranges.append({
        "name": node.name, "start": start, "end": node.end_lineno,
        "covered_statement_lines": sorted(covered),
        "missing_statement_lines": sorted(selected - executed),
        "percent": 100 * len(covered) / len(selected),
    })
owned = []
for path in [SOURCE, *TESTS]:
    raw = (ROOT / path).read_bytes()
    text = raw.decode("utf-8")
    tree = ast.parse(text)
    count = len(raw.splitlines())
    assert count <= 500
    definitions = [node for node in ast.walk(tree) if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef))]
    assert all(ast.get_docstring(node) for node in definitions)
    lines = text.splitlines()
    explicit_loops = [node for node in ast.walk(tree) if isinstance(node, (ast.For, ast.While, ast.AsyncFor))]
    assert all(lines[node.lineno - 2].lstrip().startswith("#") for node in explicit_loops)
    owned.append({**reference(ROOT / path), "physical_lines": count,
                  "documented_definitions": len(definitions), "commented_explicit_loops": len(explicit_loops)})

raw = []
for path in sorted(SCRATCH.glob("p5t3-producer-planning*")):
    if path.suffix not in {".stdout", ".stderr", ".json", ".coverage", ".py"}:
        continue
    content = path.read_bytes()
    encoded = base64.b64encode(gzip.compress(content, mtime=0)).decode("ascii")
    assert gzip.decompress(base64.b64decode(encoded)) == content
    raw.append({"identifier": "SCRATCH/" + path.name, "sha256": digest(content),
                "bytes": len(content), "encoding": "gzip+base64", "data": encoded,
                "roundtrip_verified": True})
archive = OUT / f"p5t3-producer-final-archive.{STAMP}.json"
write_json(archive, {"Timestamp": STAMP, "entries": raw,
                    "prior_raw_archive": reference(EVIDENCE / "qa-gates/p5t3-producer-source-archive.2026-10-20T09-01.json")})
harness = OUT / f"p5t3-producer-final-freeze-harness.{STAMP}.txt"
harness.write_bytes(Path(__file__).read_bytes())
manifest = OUT / f"p5t3-producer-freeze.{STAMP}.json"
value = {
    "Timestamp": STAMP, "task": "P5-T3", "state": "FOCUSED_PRODUCER_READY",
    "scope": owned, "proof": [{key: item for key, item in proof.items() if key != "coverage"} for proof in proofs],
    "coverage": {
        "convention": "Exact identical production bytes; union of raw coverage.py statements/arcs across disjoint pure and planning populations. Raw AST inclusive ranges, including executed declaration lines; no callable exclusion.",
        "statements": len(statements), "covered_statements": len(executed),
        "missing_lines": sorted(statements - executed), "branches": len(branches),
        "covered_branches": len(covered_branches), "missing_branches": sorted(branches - covered_branches),
        "callable_ranges": ranges, "production_classes": [],
    },
    "source_review_snapshot": reference(EVIDENCE / "qa-gates/p5t3-producer-source-freeze.2026-10-20T09-01.json"),
    "independent_source_review": reference(EVIDENCE / "qa-gates/p5t3-executor-producer-source-review.2026-10-20T09-09.json"),
    "raw_archive": reference(archive), "executed_assembly_harness": reference(harness),
    "qualifications": [
        "The original 35 pure cases are retained against identical producer bytes. The source review snapshot proves subsequent test documentation/comment changes preserve all executable AST, parametrizations and assertions.",
        "Planning proof binds the current semantic validator/operation dependencies before and after each stage; earlier pure proof names its historical mechanical dependencies explicitly.",
        "Two genuine in-memory loader/planner/commit compositions supplement isolated public boundary tests; no filesystem, process, network or native host grant is created by unit tests.",
        "Repository-wide gates, reversed combined population, native composition and independent acceptance remain coordinator-owned and pending.",
        "Working bytes are exact; Git normalization/index identities are separate. Raw tool outputs are losslessly encoded, including any CRLF or missing terminal newline.",
    ],
}
write_json(manifest, value)
md = manifest.with_suffix(".md")
md.write_bytes((
    f"# P5-T3 producer focused handoff\n\nTimestamp: {STAMP}\n\n"
    "Command: `poetry run python SCRATCH/p5t3-producer-final-freeze.py " + ATTEMPT + "`\n\n"
    "EXIT_CODE: 0\n\nOutput Summary: 35 retained pure cases plus 19 planning cases passed. "
    f"Measured {len(executed)}/{len(statements)} statements and {len(covered_branches)}/{len(branches)} branches; "
    f"all {len(ranges)} callable ranges meet the required threshold.\n\n"
    "The paired manifest binds exact sources, nodes, ordered receipts, raw coverage, "
    "lossless outputs and documentation-only qualifications. Task acceptance remains pending.\n"
).encode("utf-8"))
paths = [SOURCE, *TESTS]
for kind in ("baseline", "regression-testing", "qa-gates", "other"):
    paths.extend(path.relative_to(ROOT).as_posix() for path in sorted((EVIDENCE / kind).glob("p5t3-producer-*")))
stage = OUT / f"p5t3-producer-staging.{STAMP}.json"
paths.append(stage.relative_to(ROOT).as_posix())
paths = sorted(set(paths))
write_json(stage, {"Timestamp": STAMP, "manifest": reference(manifest), "stage_paths": paths,
                   "content_bindings": [reference(ROOT / path) for path in paths if path != stage.relative_to(ROOT).as_posix()]})
for path in paths:
    content = (ROOT / path).read_bytes()
    if (ROOT / path).suffix not in {".json", ".md"} or path in [SOURCE, *TESTS]:
        continue
    assert b"\r" not in content and content.endswith(b"\n") and not content.endswith(b"\n\n"), path
    assert all(line == line.rstrip() for line in content.splitlines()), path
print(json.dumps({"manifest": reference(manifest), "staging": reference(stage),
                  "cases": sum(len(proof["nodes"]) for proof in proofs),
                  "coverage": value["coverage"]}, indent=2))
