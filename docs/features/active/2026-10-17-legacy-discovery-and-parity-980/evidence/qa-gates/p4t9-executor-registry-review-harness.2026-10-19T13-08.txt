sha = lambda raw: hashlib.sha256(raw).hexdigest()


def ref(path):
    return {'path': Path(path).as_posix(), 'sha256': sha(Path(path).read_bytes())}


def load(path, digest=None):
    raw = Path(path).read_bytes()
    assert digest is None or sha(raw) == digest, str(path)
    return json.loads(raw)


def walk(value):
    if isinstance(value, dict):
        yield value
        for child in value.values():
            yield from walk(child)
    elif isinstance(value, list):
        for child in value:
            yield from walk(child)

def publish(stem, suffix, data):
    now = dt.datetime.now(dt.timezone.utc)
    stamp = now.strftime('%Y-%m-%dT%H-%M')
    path = Q / f'{stem}.{stamp}{suffix}'
    assert not path.exists()
    if isinstance(data, dict):
        data = (json.dumps({**data, 'timestamp': stamp, 'published_utc': now.isoformat()}, indent=2) + '\n').encode()
    assert b'\r' not in data and data.endswith(b'\n') and not data.endswith(b'\n\n')
    assert all(line.rstrip() == line for line in data.splitlines())
    path.write_bytes(data)
    return path
