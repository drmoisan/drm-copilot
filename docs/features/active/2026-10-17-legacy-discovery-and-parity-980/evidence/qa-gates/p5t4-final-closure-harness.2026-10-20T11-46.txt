"""Archive the completed P5-T4 run using the accepted final closure convention."""

import base64
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import sys


ROOT = Path.cwd()
SCRATCH = Path(__file__).resolve().parent
QA = ROOT / "docs/features/active/2026-10-17-legacy-discovery-and-parity-980/evidence/qa-gates"
HANDOFF = Path(sys.argv[1])
CANDIDATE = QA / "p5t4-final-candidate-released.2026-10-20T11-40.json"
STAGED = set()


def sha(data):
    """Return the identity of exact bytes, without newline conversion."""
    return hashlib.sha256(data).hexdigest()


def ref(path):
    """Bind an existing repository artifact by portable path and exact bytes."""
    path = path.resolve()
    data = path.read_bytes()
    return {"path": path.relative_to(ROOT).as_posix(), "sha256": sha(data), "bytes": len(data)}


def save(path, value):
    """Create one new LF UTF-8 receipt and retain its staging path."""
    if path.exists():
        raise ValueError("An immutable closure artifact already exists")
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n", encoding="utf-8", newline="\n")
    STAGED.add(path)
    return ref(path)


def main():
    """Preserve final raw output and current frozen bindings without executing tests."""
    stamp = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H-%M")
    handoff, candidate = json.loads(HANDOFF.read_bytes()), json.loads(CANDIDATE.read_bytes())
    assert sha(CANDIDATE.read_bytes()) == "04e475da4d094aa121bcc94fb7c703390245e182f51bf5967cbfc3f6f17aae66"
    assert handoff["status"] == "FINAL_GATES_PASSED_ACCEPTANCE_PENDING"
    assert handoff["candidate_sha256"] == sha(CANDIDATE.read_bytes())
    assert handoff["bindings"]["changed"] == [] and handoff["node_count"] == 123
    assert all(sha((ROOT / path).read_bytes()) == digest for path, digest in candidate["bindings"].items())
    STAGED.update(path for path in QA.glob("p5t4-final-*") if path.suffix in {".json", ".md", ".txt"})
    archives = []
    # Every stage already archived stdout/stderr/progress before returning.
    for record in handoff["receipts"]:
        receipt = ROOT / record["path"]
        assert sha(receipt.read_bytes()) == record["sha256"]
        payload = json.loads(receipt.read_bytes())
        archive = ROOT / payload["raw_archive"]["path"]
        assert sha(archive.read_bytes()) == payload["raw_archive"]["sha256"]
        STAGED.update((receipt, receipt.with_suffix(".md"), archive))
        archives.append(payload["raw_archive"])
    for key in ("coverage_ranges", "current_coverage"):
        record = handoff["coverage"][key]
        path = ROOT / record["path"]
        assert sha(path.read_bytes()) == record["sha256"]
        STAGED.add(path)
    run_id = handoff["receipts"][0]["path"].split("-format-", 1)[1].split(".", 1)[0]
    raw_paths = [(SCRATCH / f"p5t4-final-{run_id}-{label}{suffix}", "SCRATCH")
                 for label in ("acceptance", "combined")
                 for suffix in (".coverage", "-coverage.json")]
    raw_paths.append((SCRATCH / f"p5t4-final-{run_id}-coverage-result.json", "SCRATCH"))
    entries = []
    # Coverage databases and JSON retain exact original bytes, including missing final LF.
    for path, label in raw_paths:
        data = path.read_bytes()
        entries.append({"path": label + "/" + path.name, "bytes": len(data), "sha256": sha(data),
                        "crlf_count": data.count(b"\r\n"), "terminal_lf": data.endswith(b"\n"),
                        "base64": base64.b64encode(data).decode("ascii")})
    raw_ref = save(QA / f"p5t4-final-remaining-raw.{stamp}.json", {
        "task": "P5-T4", "timestamp": stamp, "entries": entries,
        "qualification": "Exact base64 bytes; raw databases/coverage JSON need no terminal newline. "
                         "Readable receipts are separate LF UTF-8 streams."})
    snapshot = QA / f"p5t4-final-closure-harness.{stamp}.txt"
    assert not snapshot.exists()
    snapshot.write_bytes(Path(__file__).read_bytes())
    STAGED.add(snapshot)
    ranges = json.loads((ROOT / handoff["coverage"]["coverage_ranges"]["path"]).read_bytes())
    summary = {
        "task": "P5-T4", "Timestamp": stamp,
        "Command": "poetry run python SCRATCH/close_final.py FINAL_HANDOFF", "EXIT_CODE": 0,
        "state": "FROZEN_FINAL_VERIFICATION_ACCEPTANCE_PENDING",
        "checked_commit": candidate["candidate_commit"], "candidate": ref(CANDIDATE),
        "handoff": ref(HANDOFF),
        "source_test_working_bindings": {path: candidate["bindings"][path] for path in candidate["files"]},
        "binding_qualification": "Working bytes match the released candidate and immutable owner manifests. "
                                 "Checked commit/push authority is the coordinator release; closure performs no Git operation.",
        "batch_nodes": {batch["label"]: len(batch["nodes"]) for batch in handoff["batches"]},
        "task_nodes": 123, "contract_nodes": len(handoff["contracts"]),
        "stage_seconds": {record["label"]: record["seconds"] for record in handoff["receipts"]},
        "composition": handoff["composition"], "coverage": handoff["coverage"],
        "remaining_raw_archive": raw_ref, "command_archives": archives,
        "unproved_lines": {record["path"]: record["remaining_unproved_lines"] for record in ranges["modules"]
                           if record["remaining_unproved_lines"]},
        "unproved_branches": {record["path"]: record["remaining_unproved_branches"] for record in ranges["modules"]
                              if record["remaining_unproved_branches"]},
        "retained_proof": handoff["retained_proof"], "closure_harness": ref(snapshot),
        "failures_retained": "Owner manifests retain original model and generation failures, review and repair proof.",
        "limitations": ["Task verification only; independent task acceptance remains pending.",
                        "Pure in-memory transform; no publication, CLI, installed-host or runtime-completion claim.",
                        "Scoped coverage is not a newly measured repository-wide coverage result."],
        "qualification": "Final fresh coverage retains each focused executed line/arc. "
                         "Historical generation 49+4 proofs remain separate from this complete 123-node process. "
                         "No source/test edit, behavioral rerun or acceptance checkoff occurred during closure.",
    }
    closure = QA / f"p5t4-final-closure.{stamp}.json"
    save(closure, summary)
    markdown = QA / f"p5t4-final-closure.{stamp}.md"
    markdown.write_text(
        "# P5-T4 final verification\n\n"
        f"Timestamp: {stamp}\nCommand: poetry run python SCRATCH/close_final.py FINAL_HANDOFF\nEXIT_CODE: 0\n\n"
        f"All seven stages passed on `{candidate['candidate_commit']}`. "
        "One fresh process executed the reversed 123-case union. Four applicable repository contracts passed. "
        "Nine existing in-memory composition controls are included in the union.\n\n"
        "Three production modules and every class/callable range meet the prepared coverage gates, "
        "with no lost focused statements or branch arcs. Remaining raw gaps are explicitly retained. "
        "Source/test bytes remained frozen; earlier owner failures and repair evidence remain referenced.\n\n"
        f"Final handoff: `{HANDOFF.name}`; SHA-256 `{sha(HANDOFF.read_bytes())}`.\n\n"
        "Raw databases, coverage JSON and command streams are preserved without newline conversion. "
        "This proves the scoped pure transform on this Windows environment. It does not prove publication, "
        "CLI behavior, installed-host hooks or runtime acceptance of a migrated application. "
        "Independent task acceptance remains pending.\n",
        encoding="utf-8", newline="\n")
    STAGED.add(markdown)
    # Check authored/readable streams independently from embedded lossless raw payloads.
    for path in STAGED:
        raw = path.read_bytes()
        text = raw.decode("utf-8")
        assert b"\r" not in raw and raw.endswith(b"\n") and not raw.endswith(b"\n\n"), path.name
        assert all(line == line.rstrip() for line in text.splitlines()), path.name
        assert str(ROOT) not in text and ROOT.as_posix() not in text, path.name
        assert str(SCRATCH) not in text and SCRATCH.as_posix() not in text, path.name
    staging = QA / f"p5t4-final-staging.{stamp}.json"
    save(staging, {"task": "P5-T4", "timestamp": stamp, "closure": ref(closure),
                  "stage_paths": sorted(path.relative_to(ROOT).as_posix() for path in STAGED | {staging}),
                  "bindings": [ref(path) for path in sorted(STAGED)],
                  "self_qualification": "Self path is listed; exact SHA is reported externally."})
    print(json.dumps({"closure": ref(closure), "staging": ref(staging),
                      "stage_path_count": len(STAGED), "raw_entries": len(entries)}))


if __name__ == "__main__":
    main()
