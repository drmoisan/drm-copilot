def bound(path, expected):
    """Read one immutable prior receipt only when its exact recorded digest matches."""
    raw = path.read_bytes()
    assert hashlib.sha256(raw).hexdigest() == expected, path
    return json.loads(raw), {"path": path.as_posix(), "sha256": expected}


def save_json(path, payload):
    """Write a new canonical LF UTF-8 document with no replacement or blank padding."""
    assert not path.exists(), path
    path.write_text(json.dumps(payload, indent=2, sort_keys=True) + "\n", encoding="utf-8", newline="\n")
