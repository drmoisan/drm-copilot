"""Publish the accepted P4-T8 checkoff without replaying reconciled proof."""
from pathlib import Path
import datetime as dt
import hashlib
import json
import re
import subprocess

F = Path('docs/features/active/2026-10-17-legacy-discovery-and-parity-980')
Q = F / 'evidence/qa-gates'
sha = lambda data: hashlib.sha256(data).hexdigest()


def ref(path):
    return {'path': Path(path).as_posix(), 'sha256': sha(Path(path).read_bytes())}


def publish(stem, suffix, data):
    now = dt.datetime.now(dt.timezone.utc)
    stamp = now.strftime('%Y-%m-%dT%H-%M')
    path = Q / f'{stem}.{stamp}{suffix}'
    assert not path.exists(), str(path)
    if isinstance(data, dict):
        data = (json.dumps({**data, 'timestamp': stamp, 'published_utc': now.isoformat()}, indent=2) + '\n').encode()
    assert b'\r' not in data and data.endswith(b'\n') and not data.endswith(b'\n\n')
    assert all(line.rstrip() == line for line in data.splitlines())
    path.write_bytes(data)
    return path


reconciliation = Q / 'p4t8-executor-final-reconciliation.2026-10-19T12-31.json'
assert sha(reconciliation.read_bytes()) == '2e4b875be81b4dc7576ca6e618b74fc243633b79fe6a168a0557e95729f4c61f'
proof = json.loads(reconciliation.read_bytes())
assert proof['status'] == 'FINAL_RECONCILIATION_PASS' and proof['findings'] == []
plan_path = Path(proof['plan_before']['path'])
prior = subprocess.check_output(['git', 'show', 'HEAD:' + plan_path.as_posix()])
assert sha(prior) == proof['plan_before']['sha256']
expected = prior.replace(b'- [ ] [P4-T8]', b'- [x] [P4-T8]', 1)
assert expected != prior and plan_path.read_bytes() == expected
tasks = re.findall(rb'^- \[([ x])\] \[(P\d+-T\d+)\]', expected, re.M)
assert len(tasks) == 84 and sum(flag == b'x' for flag, _ in tasks) == 40
for item in proof['ac_mapping']['sources']:
    assert sha(Path(item['path']).read_bytes()) == item['sha256']
command = 'poetry run python SCRATCH/issue-980-p4t8-final-acceptance.py'
result = {'task_status': 'PASS', 'task': 'P4-T8', 'command': command, 'exit_code': 0,
          'reconciliation': ref(reconciliation), 'handoff': proof['handoff'], 'retained_review': proof['retained_review'],
          'plan': ref(plan_path), 'plan_checkoff': 'P4-T8 only; exact one-checkbox byte delta', 'plan_count': '40/84',
          'source_ac_count': '13/66 in each authoritative source; unchanged', 'new_source_ac_checkoffs': [],

          'limitations': proof['retained_limitations'], 'findings': [], 'new_behavioral_runs': 0,
          'next_five_tasks': ['P4-T9', 'P5-T1', 'P5-T2', 'P5-T3', 'P5-T4'],
          'source_test_checkpoint_git_mutations': False}
accepted = publish('p4t8-executor-final-acceptance', '.json', result)

    'Output Summary: TASK_STATUS: PASS; TASK: P4-T8. Only the P4-T8 plan checkbox changed.\n\n'
    'The retained candidate review and final clean ordered sequence agree on the same committed bytes.                   'counts': ['40/84', '13/66']}))
