def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def reference(path):
    return {"path": path.as_posix(), "sha256": sha(path.read_bytes())}


def archive(raw):
    return {"sha256": sha(raw), "base64": base64.b64encode(raw).decode("ascii")}


def write_json(path, value):
    path.write_text(
        json.dumps(value, indent=2, sort_keys=True) + "\n",
        encoding="utf-8", newline="\n",
    )


def git_bytes(path):
    return subprocess.check_output(["git", "show", "HEAD:" + path])


def normalize_text(raw):
    return raw.replace(b"\r\n", b"\n")


def body(node):
    statements = node.body
    if statements and isinstance(statements[0], ast.Expr):
        value = statements[0].value
        if isinstance(value, ast.Constant) and isinstance(value.value, str):
            statements = statements[1:]
    return ast.dump(ast.Module(body=statements, type_ignores=[]), include_attributes=False)

    assert all(
        ast.get_docstring(node)
        for node in ast.walk(tree)
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef))
    )
    bindings[path] = {
        "working_sha256": sha(raw), "physical_lines": len(lines),
        "working_git_blob": subprocess.check_output(
            ["git", "hash-object", "--no-filters", path]
        ).decode().strip(),
        "attributes": subprocess.check_output(
            ["git", "check-attr", "text", "eol", "--", path]
        ).decode().strip().splitlines(),
    }

    tree = ast.parse(Path(path).read_text(encoding="utf-8"))
    for node in ast.walk(tree):
        if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            continue
        executed = [line for line in data["executed_lines"] if node.lineno <= line <= node.end_lineno]
        missing = [line for line in data["missing_lines"] if node.lineno <= line <= node.end_lineno]
        branches = [edge for edge in data["executed_branches"] if node.lineno <= edge[0] <= node.end_lineno]
        absent = [edge for edge in data["missing_branches"] if node.lineno <= edge[0] <= node.end_lineno]

            assert len(executed) / (len(executed) + len(missing)) >= 0.90
        units.append({
            "path": path, "name": node.name, "kind": type(node).__name__,
            "start": node.lineno, "end": node.end_lineno,
            "executed_lines": executed, "missing_lines": missing,
            "executed_branches": branches, "missing_branches": absent,
        })
