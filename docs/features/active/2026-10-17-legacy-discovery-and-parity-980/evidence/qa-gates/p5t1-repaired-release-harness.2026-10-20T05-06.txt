"""Bind the released checked repair to the existing prepared gate sequence."""
import datetime
import hashlib
import json
from pathlib import Path
import subprocess
import sys

scratch = Path(__file__).resolve().parent
qa = Path("docs/features/active/2026-10-17-legacy-discovery-and-parity-980/evidence/qa-gates")
preparation_path = qa / "p5t1-repaired-final-preparation.2026-10-20T04-58.json"


def sha(data):
    """Return exact byte identity without normalization."""
    return hashlib.sha256(data).hexdigest()


def ref(path):
    """Return a portable reference to one existing evidence stream."""
    return {"path": path.as_posix(), "sha256": sha(path.read_bytes())}


def write(path, value):
    """Create a new canonical LF receipt with one terminal newline."""
    assert not path.exists(), path
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n", encoding="utf-8", newline="\n")


assert sha(preparation_path.read_bytes()) == "e6ae92826e1dd8e6916ee2dad7b03ed9ce3bc6152312f6929d78bb67b52e0f05"
manifest = json.loads(preparation_path.read_bytes())
state = json.loads(Path("artifacts/orchestration/orchestrator-state.json").read_bytes())
route = next(row for row in state["codex_model_routing_receipts"]
             if row["delegation_id"] == "workspace_operations_p5t1_repaired_final_gates_20261020")
assert route["profile_sha256"] == sha(Path(route["profile_path"]).read_bytes()) == "0d02ca1f9b97abd4d84120f9443359404ab6ef4d094a1b97d520381fb1f9edd6"
authority_path = Path.home() / ".codex/authority/epic-entry/0a00160052dfcee84f12df9a9b8cfcdde1c2dedd8ace3ebd457a161fa9c1b2c3/01a15153-48ad-74d1-9f2d-e3ad413dafb4/codex-routing-attestation.f50a381e7b3554fe77176b5c0429a53b56579adb36427614259db5bc6770a5d6.json"
authority = json.loads(authority_path.read_bytes())
assert all(authority[key] is True for key in ("routing_valid", "launch_authority_valid", "provenance_valid"))
assert authority["profile_sha256"] == route["profile_sha256"]
head = subprocess.run(["git", "rev-parse", "HEAD"], capture_output=True, check=True).stdout.decode().strip()
assert head == "0f30c498300e69b231d51a1baaa88e9c061dd132"
confirmation = qa / "p5t1-executor-output-scope-confirmation.2026-10-20T05-02.json"
assert sha(confirmation.read_bytes()) == "559a1efe40646822281dea9bdbc81c54c60064fcc06d3e60cc17f47ea1afd3aa"
for path, expected in manifest["bindings"].items():
    assert sha(Path(path).read_bytes()) == expected, path
for name, entry in manifest["harnesses"].items():
    assert sha((scratch / name).read_bytes()) == entry["sha256"], name
for entry in manifest["files"]:
    path = entry["path"]
    raw = Path(path).read_bytes()
    committed = subprocess.run(["git", "show", head + ":" + path], capture_output=True, check=True).stdout
    assert raw == committed or raw.replace(b"\r\n", b"\n") == committed, path
    entry.update(git_sha256=sha(committed),
                 git_blob=subprocess.run(["git", "rev-parse", head + ":" + path], capture_output=True, check=True).stdout.decode().strip(),
                 normalization="exact" if raw == committed else "CRLF-to-LF only",
                 git_binding_status="Checked repair HEAD verified")
stamp = datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H-%M")
harness = qa / ("p5t1-repaired-release-harness." + stamp + ".txt")
assert not harness.exists()
harness.write_bytes(Path(__file__).read_bytes())
manifest["harnesses"][Path(__file__).name] = ref(harness)
manifest.update(execution_released=True, route=route, timestamp=stamp, head=head,
                preparation=ref(preparation_path), repair_confirmation=ref(confirmation),
                authority_sha256=sha(authority_path.read_bytes()),
                repair_confirmation_and_checked_commit="Verified exact confirmation and checked repair HEAD.")
candidate = qa / ("p5t1-repaired-final-candidate." + stamp + ".json")
write(candidate, manifest)
print(json.dumps({"candidate": ref(candidate), "head": head, "bindings": len(manifest["bindings"]),
                  "sequence": manifest["new_sequence"], "started": True}), flush=True)
raise SystemExit(subprocess.call([sys.executable, "-u", str(scratch / "issue-980-p5t1-repaired-gates.py"), candidate.as_posix()]))
