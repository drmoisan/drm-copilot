sha = lambda raw: hashlib.sha256(raw).hexdigest()


def load(path, digest=None):
    raw = Path(path).read_bytes()
    assert digest is None or sha(raw) == digest, str(path)
    return json.loads(raw)


def ref(path):
    return {'path': Path(path).as_posix(), 'sha256': sha(Path(path).read_bytes())}


def walk(value):
    if isinstance(value, dict):
        yield value
        for child in value.values():
            yield from walk(child)
    elif isinstance(value, list):
        for child in value:
            yield from walk(child)


def publish(stem, suffix, value):
    now = dt.datetime.now(dt.timezone.utc)
    path = Q / (stem + '.' + now.strftime('%Y-%m-%dT%H-%M') + suffix)
    assert not path.exists()
    raw = ((json.dumps({**value, 'published_utc': now.isoformat()}, indent=2) + '\n').encode() if isinstance(value, dict) else value)
    assert b'\r' not in raw and raw.endswith(b'\n') and not raw.endswith(b'\n\n')
    assert all(line == line.rstrip() for line in raw.splitlines())
    path.write_bytes(raw)
    return path
