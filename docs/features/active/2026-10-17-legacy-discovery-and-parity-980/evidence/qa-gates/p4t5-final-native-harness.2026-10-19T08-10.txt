import yaml
from drm_copilot_discovery.analyzer.com_interop import ComInteropAnalyzer

from drm_copilot_discovery.analyzer.registry_access import RegistryAccessAnalyzer

scratch=Path(__file__).resolve().parent
root=scratch/('native-api-interop-'+uuid4().hex)
root.mkdir()
assert root.resolve().parent==scratch
checks=[]

def check(name,value):
    if not value: raise AssertionError(name)
    checks.append(name)
    print('PASS '+name,flush=True)

def put(path,content):
    destination=root/path
    assert root in destination.resolve().parents
    destination.parent.mkdir(parents=True,exist_ok=True)
    destination.write_bytes(content)

def tree():
    return {path.relative_to(root).as_posix():hashlib.sha256(path.read_bytes()).hexdigest() for path in sorted(root.rglob('*')) if path.is_file() and '.git' not in path.relative_to(root).parts}

def git(*arguments):
    return subprocess.run(['git','-C',str(root),*arguments],capture_output=True,check=True).stdout

def run_git(argv,*,output_limit):
    process=subprocess.Popen(argv,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
    try:
        content=process.stdout.read(output_limit+1)
        if len(content)>output_limit:
            process.kill()
            raise OSError('Native Git output exceeded its bounded read')
        if process.wait()!=0: raise OSError('Native Git read failed')
        return content
    finally:
        if process.poll() is None: process.kill()
        process.wait(); process.stdout.close()

put('src/a.cs',b'File.Open("VALUE_SENTINEL_PATH");\r\nRegistry.CurrentUser.OpenSubKey("VALUE_SENTINEL_KEY");\r\nMarshal.ReleaseComObject(value);\r\n')
put('src/b.csproj',b'<Project><ItemGroup><COMReference/></ItemGroup></Project>')
put('.gitattributes',b'src/* -text\n')
git('init','--quiet'); git('config','core.autocrlf','false'); git('add','src','.gitattributes')
git('-c','user.name=Discovery fixture','-c','user.email=discovery@example.invalid','-c','commit.gpgsign=false','commit','--quiet','-m','Synthetic API and interop inventory')
revision=git('rev-parse','HEAD').decode('ascii').strip()
catalog=load_catalog()
value=yaml.safe_load(Path('examples/discovery/v2/domain-profile.example.yaml').read_text(encoding='utf-8'))
value.pop('extension_schemas'); value['system']['role']='legacy-source'
value['repository']['revision']=revision; value['repositories']=[]; value['clients']=[]
value['production_roots']=['src']; value['test_roots']=[]; value['documentation_roots']=[]
value['project_entry_points']=[]; value['excluded_paths']=[]
identifiers=('repository','filesystem-access','registry-access','com-interop')
value['analyzers']={'enabled':list(identifiers),'options':{'max_file_bytes':16384},'ignored_paths':[],'extension_result_manifests':[]}
profile=ProfileV2(json.dumps(value),format='json',catalog=catalog)
reader=LocalGitReader(workspace_root=str(root),repository_id=profile.repository.id,run=run_git)
primitives=LocalMutationPrimitives(workspace_root=str(root),lock_path=str(scratch/('native-api-interop-'+uuid4().hex+'.lock')))
before=tree(); status=git('status','--porcelain=v1','-z')
inputs=capture_inputs(profile,workspace_root=str(root),metadata=primitives.metadata,read_bytes=primitives.read_bytes)
check('native capture retains both source and project inputs',len(inputs.inputs)==2 and all(item.file is not None for item in inputs.inputs))
pins=tuple(reader.read_file(revision=revision,path=item.path,limit=16384) for item in inputs.inputs)
inputs=replace(inputs,repository_files=pins)
check('real Git identity and captured bytes agree',reader.identity().revision==revision and all(pin.content==item.file.content for pin,item in zip(pins,inputs.inputs)))
filesystem=FilesystemAccessAnalyzer().analyze(inputs)
registry=RegistryAccessAnalyzer().analyze(inputs)
com=ComInteropAnalyzer().analyze(inputs)
check('filesystem syntax remains one inferred pinned candidate',len(filesystem.findings)==1 and filesystem.findings[0]['classification']=='inferred' and filesystem.findings[0]['confidence']=='medium')
check('registry syntax leaves platform applicability unresolved',len(registry.findings)==1 and registry.findings[0]['classification']=='inferred' and registry.findings[0]['extensions']['drm.api-access']['platform_applicability']=='unresolved')
check('COM declarations distinguish lexical calls from project facts',len(com.findings)==2 and {item['kind']:item['classification'] for item in com.findings}=={'com_lifetime_call':'inferred','com_reference':'detected'} and all(item['extensions']['drm.com-interop']['execution']=='unverified' for item in com.findings))
analyses=(filesystem,registry,com)
check('all exact pinned adapters remain free of blocking diagnostics',not any(item['blocking'] for analysis in analyses for item in analysis.diagnostics))
encoded=canonical_json(row(results=tuple(row(components=analysis.components,findings=analysis.findings,relationships=analysis.relationships,diagnostics=analysis.diagnostics) for analysis in analyses)))
check('runtime paths keys and argument values never enter results',b'VALUE_SENTINEL' not in encoded)
check('each source reference retains its exact captured hash and revision',all(item['source']['sha256']==before[item['source']['path']] and item['source']['revision']==revision for analysis in analyses for item in analysis.components+analysis.findings))
structural=RepositoryAnalyzer().analyze(inputs)
result=inventory_result(inputs,(structural,*analyses),analyzer_ids=identifiers,timestamp='2026-10-19T00:00:00Z',author_or_agent=row(kind='tool',id='native-proof'),catalog=catalog)
repeated=inventory_result(inputs,(*reversed(analyses),structural),analyzer_ids=tuple(reversed(identifiers)),timestamp='2026-10-20T00:00:00Z',author_or_agent=row(kind='human',id='second-observer'),catalog=catalog)
check('composed identity is independent of adapter order and capture clock',result.semantic_sha256==repeated.semantic_sha256 and result.document['id']==repeated.document['id'])
check('ledger proposals remain uninspected and unassigned',all(item['inspection_status']=='uninspected' and item['disposition']=='unassigned' for item in result.ledger_components))
check('analysis preserves native source tree and Git status',before==tree() and status==git('status','--porcelain=v1','-z'))
output=Path(sys.argv[1])
record={'task':'P4-T5','timestamp':datetime.datetime.now(datetime.timezone.utc).isoformat(),'platform':sys.platform,'checks':checks,'scratch_repository':'SCRATCH/'+root.name,'revision':revision,'profile':value,'inputs':before,'inventory_id':result.document['id'],'semantic_sha256':result.semantic_sha256,'component_count':len(result.document['components']),'finding_count':len(result.document['findings']),'relationship_count':len(result.document['relationships']),'source_tree_unchanged':before==tree(),'git_status_unchanged':status==git('status','--porcelain=v1','-z')}
output.write_text(json.dumps(record,indent=2,sort_keys=True)+'\n',encoding='utf-8',newline='\n')
print(output.as_posix(),flush=True)
