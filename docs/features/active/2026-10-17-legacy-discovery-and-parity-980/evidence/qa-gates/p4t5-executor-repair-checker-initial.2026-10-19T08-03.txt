"""Bind the bounded P4-T5 repair confirmation without replaying any unit suite."""
from datetime import datetime, timezone
from hashlib import sha256
import json
from pathlib import Path
import subprocess

FEATURE = Path("docs/features/active/2026-10-17-legacy-discovery-and-parity-980")
E = FEATURE / "evidence"
Q = E / "qa-gates"
START = datetime.now(timezone.utc)


def digest(data):
    return sha256(data).hexdigest()


def ref(path):
    return {"path": Path(path).as_posix(), "sha256": digest(Path(path).read_bytes())}


def load(path):
    return json.loads(Path(path).read_bytes())


def verify_refs(value):
    if isinstance(value, dict):
        if "path" in value and "sha256" in value and value["path"].startswith("docs/"):
            assert ref(value["path"])["sha256"] == value["sha256"], value["path"]
        for item in value.values():
            verify_refs(item)
    elif isinstance(value, list):
        for item in value:
            verify_refs(item)


def source_map(value):
    return {p: r["sha256"] if isinstance(r, dict) else r for p, r in value.items()}


paths = [
    Q / "p4t5-access-repair-freeze.2026-10-19T07-56.json",
    E / "regression-testing/p4t5-com-repair-freeze.2026-10-19T07-58.json",
    Q / "p4t5-executor-repair-probes.2026-10-19T07-59.json",
    Q / "p4t5-executor-candidate-review.2026-10-19T07-42.json",
]
expected = ["ac19d12cb7096d9c30e86470cff02032a9d1ac87055db3c39292f875bc76b3c4",
            "d3914889c412fd3c0e81d9aaedb4284328dcda050a47a943b9800f3fa1eebfc4",
            "cc6986519a4eb11a7350967379f2a72e90a06c6ee77ddd77efc4a02cf7d111a5",
            "95773ccf65e1e9f7891338871b4699e13ac51515979bc57943389d4e3fd71726"]
for path, expected_hash in zip(paths, expected, strict=True):
    assert ref(path)["sha256"] == expected_hash
access, com, probes, review = [load(path) for path in paths]
for document in (access, com, probes):
    verify_refs(document)
assert probes["failed_expectations"] == 0 and probes["case_count"] == 12
assert ref(Q / "p4t5-executor-candidate-review.2026-10-19T07-42.md")["sha256"] == "364c0ab5d53e2f70ef4ccebe53b5fa207cd0c13bf0f8ee75bf2a11a50a61e881"

source_rows = []
for row in access["files"] + com["source_files"]:
    path = row["path"]
    data = Path(path).read_bytes()
    assert digest(data) == row["sha256"] and b"\r" not in data
    assert len(data.splitlines()) <= 500
    index = subprocess.run(["git", "show", ":" + path], capture_output=True, check=False)
    source_rows.append({"path": path, "working_sha256": digest(data), "lines": len(data.splitlines()),
                        "index_sha256": digest(index.stdout) if index.returncode == 0 else None,
                        "working_equals_index": index.returncode == 0 and index.stdout == data,
                        "qualification": "Working bytes are the reviewed repair; the coordinator owns staging and commit."})
assert len(source_rows) == 10
gate_rows = []
for label, items, files, count in (("access", access["ordered_receipts"], access["files"], 180),
                                  ("COM", com["ordered_focused_gates"], com["source_files"], 137)):
    expected_sources = {row["path"]: row["sha256"] for row in files}
    last = None
    for item in items:
        gate = load(item["path"])
        assert gate["exit_code"] == 0
        assert source_map(gate["source_before"]) == expected_sources == source_map(gate["source_after"])
        start = datetime.fromisoformat(gate.get("started_at", gate.get("started")))
        finish = datetime.fromisoformat(gate.get("finished_at", gate.get("finished")))
        assert start <= finish and (last is None or last <= start)
        last = finish
        if item["stage"] in {"unit", "tests"}:
            assert str(count) + " passed" in gate.get("stdout", gate.get("output", ""))
        gate_rows.append({"subset": label, "stage": item["stage"], "exit_code": 0, "receipt": ref(item["path"])})

ranges = load(access["coverage_ranges"]["path"])

coverage_access = load(ranges["coverage"]["path"])
coverage_com = load(com["coverage"]["path"])

assert coverage_access["totals"]["covered_branches"] == coverage_access["totals"]["num_branches"] == 94
assert coverage_com["totals"]["covered_lines"] == coverage_com["totals"]["num_statements"] == 206
assert coverage_com["totals"]["covered_branches"] == coverage_com["totals"]["num_branches"] == 88
for unit in ranges["units"]:
    assert unit["covered_statements"] == unit["statements"] and unit["covered_branches"] == unit["branches"]
com_structure = load(com["coverage_structure"]["path"])
com_units = 0
for module in com_structure["summaries"]:

        for summary in group.values():
            assert summary["covered_lines"] == summary["num_statements"]
            assert summary["covered_branches"] == summary["num_branches"]
            com_units += 1
assert len(ranges["units"]) == 15 and com_units == 12
