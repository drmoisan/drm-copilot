def sha(data):
    """Bind exact bytes without normalization."""
    return hashlib.sha256(data).hexdigest()


def reference(relative, expected):
    """Verify an existing canonical artifact and retain its portable identity."""
    path = FEATURE / relative
    assert sha(path.read_bytes()) == expected, relative
    return {"path": path.as_posix(), "sha256": expected}


def records(value):
    """Find the exact routing record without replicating checkpoint contents."""
    if isinstance(value, dict):
        yield value
        # Existing checkpoint receipts occur in both mappings and arrays.
        for item in value.values():
            yield from records(item)
    elif isinstance(value, list):
        # Receipt identity, rather than an array position, determines authority.
        for item in value:
            yield from records(item)


def write_json(path, value):
    """Create one LF UTF-8 receipt without replacing retained evidence."""
    assert not path.exists(), path
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n",
                    encoding="utf-8", newline="\n")
    return {"path": path.as_posix(), "sha256": sha(path.read_bytes())}
