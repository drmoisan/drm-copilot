def sha(data):
    """Bind exact bytes without newline or encoding normalization."""
    return hashlib.sha256(data).hexdigest()


def reference(relative, expected):
    """Verify one existing canonical artifact without rewriting its bytes."""
    path = FEATURE / relative
    data = path.read_bytes()
    assert sha(data) == expected, relative
    return {"path": path.as_posix(), "sha256": expected}


def records(value):
    """Find the existing routing record without copying the whole checkpoint."""
    if isinstance(value, dict):
        yield value
        # The checkpoint contains both list and mapping receipt containers.
        for item in value.values():
            yield from records(item)
    elif isinstance(value, list):
        # Preserve receipt identity rather than assuming a checkpoint array index.
        for item in value:
            yield from records(item)


def write_json(path, value):
    """Write a new LF UTF-8 evidence file and refuse historical replacement."""
    assert not path.exists(), path
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n",
                    encoding="utf-8", newline="\n")
    return {"path": path.as_posix(), "sha256": sha(path.read_bytes())}
