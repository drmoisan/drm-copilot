"""Freeze FR1/FR3 proof while checking original evidence and tests remain intact."""
import ast
import datetime
import hashlib
import json
from pathlib import Path
import re

repo, scratch = Path.cwd(), Path(__file__).parent
evidence = repo / "docs/features/active/2026-10-17-legacy-discovery-and-parity-980/evidence/regression-testing"
stamp = datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H-%M")
owned = ["packages/discovery-runtime/src/drm_copilot_discovery/analyzer/dependency_python.py", "tests/discovery_runtime/analyzer/test_dependency_python.py"]
shared = {
    "packages/discovery-runtime/src/drm_copilot_discovery/analyzer/dependency_inputs.py": "9bf1b6e01a4adac30bb8481a88fc3a70cff8b66af5abfdb712b5908100749a90",
    "packages/discovery-runtime/src/drm_copilot_discovery/analyzer/dependency_values.py": "3b2c5c1024e941839cdbd60701a3f07cff4fd78ab0c1a47a1b20de1e73450c3a",
}


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def bind(path):
    source = repo / path
    return {"sha256": digest(source), "physical_lines": len(source.read_text("utf-8").splitlines())}


bindings = {path: bind(path) for path in owned + list(shared)}
assert all(item["physical_lines"] <= 500 for item in bindings.values())
assert all(bindings[path]["sha256"] == expected for path, expected in shared.items())
baseline_path = evidence / "p4t7-python-freeze.2026-10-19T10-12.json"
assert digest(baseline_path) == "1ec6826e8b621ff41bf5a38eb959a179541bf802e3cacd740bfff6915a08259a"
baseline = json.loads(baseline_path.read_text("utf-8"))
assert all(digest(repo / path) == expected for path, expected in baseline["artifacts"].items())

# Compare complete old test bodies and parametrizations after removing only docstrings.
old_tree = ast.parse((evidence / "p4t7-python-full-file-2.2026-10-19T10-12.txt").read_text("utf-8"))
new_tree = ast.parse((repo / owned[1]).read_text("utf-8"))


def functions(tree):
    result = {}
    for node in tree.body:
        if isinstance(node, ast.FunctionDef):
            if ast.get_docstring(node) is not None:
                node.body = node.body[1:]
            result[node.name] = ast.dump(node, include_attributes=False)
    return result


old_functions, new_functions = functions(old_tree), functions(new_tree)
assert all(new_functions[name] == body for name, body in old_functions.items())
records = sorted(evidence.glob("p4t7-python-repair-focused2-*.json"))
assert len(records) == 4
for path in records:
    record = json.loads(path.read_text("utf-8"))
    assert record["exit_code"] == 0
    assert record["source_before"] == record["source_after"] == bindings
    for stream in record["streams"].values():
        assert digest(scratch / stream["scratch_identifier"].removeprefix("SCRATCH/")) == stream["raw_sha256"]
unit = json.loads(next(evidence.glob("p4t7-python-repair-focused2-unit.*.json")).read_text("utf-8"))
nodes = re.findall(r"^(tests/.*) PASSED", unit["streams"]["stdout"]["presentation"], re.M)
assert len(nodes) == 116 and set(baseline["nodes"]) <= set(nodes)
coverage = json.loads((scratch / "issue-980-p4t7-python-repair-coverage.json").read_text("utf-8"))
covered = next(iter(coverage["files"].values()))
assert covered["summary"]["missing_lines"] == covered["summary"]["missing_branches"] == 0
for collection in ("functions", "classes"):
    assert all(item["summary"]["missing_lines"] == item["summary"]["missing_branches"] == 0 for item in covered.get(collection, {}).values())
(evidence / f"p4t7-python-repair-coverage.{stamp}.json").write_bytes((json.dumps(coverage, indent=2) + "\n").encode())
for suffix in ("gate", "freeze"):
    (evidence / f"p4t7-python-repair-{suffix}-harness.{stamp}.txt").write_bytes((scratch / f"issue-980-p4t7-python-repair-{suffix}.py").read_bytes())
for index, path in enumerate(owned, 1):
    (evidence / f"p4t7-python-repair-full-file-{index}.{stamp}.txt").write_bytes((repo / path).read_bytes())
