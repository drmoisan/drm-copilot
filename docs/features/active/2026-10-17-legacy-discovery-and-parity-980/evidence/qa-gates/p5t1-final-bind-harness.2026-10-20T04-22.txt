def digest(data):
    """Return SHA-256 for exact supplied bytes."""
    return hashlib.sha256(data).hexdigest()


def ref(path):
    """Bind one existing portable path to its actual bytes."""
    return {"path": path.as_posix(), "sha256": digest(path.read_bytes())}


def save(path, value):
    """Create portable LF JSON without replacing previous evidence."""
    assert not path.exists(), path
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n", encoding="utf-8", newline="\n")


def read(path, expected=None):
    """Read a JSON receipt and optionally assert its immutable identity."""
    data = path.read_bytes()
    assert expected is None or digest(data) == expected, path
    return json.loads(data)
