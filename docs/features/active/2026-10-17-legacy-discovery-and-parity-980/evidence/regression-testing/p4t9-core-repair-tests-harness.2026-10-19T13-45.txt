chunks, inventory = [], []

    path = test_root / filename
    text = path.read_text(encoding="utf-8")
    lines = text.splitlines(keepends=True)
    ranges = []
    for node in ast.parse(text).body:
        if isinstance(node, ast.FunctionDef) and node.name in names:
            start = min([node.lineno] + [item.lineno for item in node.decorator_list]) - 1
            end = node.end_lineno
            chunk = "".join(lines[start:end])
            chunks.append(chunk.rstrip())
            inventory.append({"name": node.name, "from": filename, "to": new_name,

            ranges.append((start, end))
    assert len(ranges) == len(names)
    for start, end in sorted(ranges, reverse=True):
        del lines[start:end]
    text = "".join(lines)
    while "\n\n\n\n" in text:
        text = text.replace("\n\n\n\n", "\n\n\n")

for filename in (*move, new_name):
    path = test_root / filename
    text = path.read_text(encoding="utf-8")
    lines = text.splitlines(keepends=True)
    replacements = []
    for node in ast.walk(ast.parse(text)):
        if isinstance(node, ast.FunctionDef):
            first = node.body[0]
            assert isinstance(first, ast.Expr) and isinstance(first.value, ast.Constant)
            assert node.name in docs, node.name
            indent = " " * first.col_offset
            content = docs[node.name].splitlines()
            rendered = indent + '"""' + content[0]
            if len(content) == 1:
                rendered += '"""\n'
            else:
                rendered += "\n" + "\n".join(indent + line[4:] if line.startswith("    ") else line for line in content[1:])
                rendered += "\n" + indent + '"""\n'
            replacements.append((first.lineno - 1, first.end_lineno, rendered))
    for start, end, rendered in sorted(replacements, reverse=True):
        lines[start:end] = [rendered]
    text = "".join(lines)
    text = text.replace('    limits = as_object(to_plain_value(bundle.profile.document["limits"]))\n    limits.update(source_files=1, artifact_records=1)\n    profile = inventory_profile(bundle.catalog, limits=limits)',
