state = json.loads(Path('artifacts/orchestration/orchestrator-state.json').read_bytes())


def routes(value):
    """Select the exact persisted delegation receipt without checkpoint mutation."""
    if isinstance(value, dict):
        if value.get('delegation_id') == 'workspace_operations_p4t9_final_gates_restart':
            yield value
        for item in value.values():
            yield from routes(item)
    elif isinstance(value, list):
        for item in value:
            yield from routes(item)
