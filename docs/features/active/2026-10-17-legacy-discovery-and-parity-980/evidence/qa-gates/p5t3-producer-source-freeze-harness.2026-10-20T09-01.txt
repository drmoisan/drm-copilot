def digest(raw: bytes) -> str:
    """Return exact byte identity."""
    return hashlib.sha256(raw).hexdigest()


class Executable(ast.NodeTransformer):
    """Remove only docstrings from an already parsed test module."""

    def generic_visit(self, node: ast.AST) -> ast.AST:
        """Retain all executable statements, arguments, decorators and assertions."""
        super().generic_visit(node)
        if isinstance(node, (ast.Module, ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
            if node.body and isinstance(node.body[0], ast.Expr) and isinstance(node.body[0].value, ast.Constant) and isinstance(node.body[0].value.value, str):
                node.body.pop(0)
        return node


def executable(raw: bytes) -> str:
    """Return a normalized AST without source-location or documentation fields."""
    return ast.dump(Executable().visit(ast.parse(raw.decode("utf-8"))), include_attributes=False)


def record(path: Path) -> dict[str, object]:
    """Describe one exact source or evidence stream."""
    raw = path.read_bytes()
    return {"path": path.relative_to(ROOT).as_posix(), "sha256": digest(raw), "bytes": len(raw)}


files = []
for path in SCOPE:
    value = record(ROOT / path)
    value["physical_lines"] = len((ROOT / path).read_bytes().splitlines())
    assert value["physical_lines"] <= 500
    files.append(value)

qualifications = []
for current, original in (
    (SCOPE[1], "p5t3-producer-pure2-original-test.py"),
    (SCOPE[2], "p5t3-producer-planning2-original-test.py"),
):
    old, new = (SCRATCH / original).read_bytes(), (ROOT / current).read_bytes()
    identical = executable(old) == executable(new)
    assert identical
    qualifications.append({
        "path": current, "old_sha256": digest(old), "current_sha256": digest(new),
        "qualification": "Test docstrings/comments only; executable AST, decorators, parametrizations and assertions identical.",
        "executable_ast_identical": identical,
    })

raw_paths = [
    SCRATCH / "p5t3-producer-focused.py",
    SCRATCH / "p5t3-producer-planning-focused.py",
    SCRATCH / "p5t3-producer-pure2-original-test.py",
    SCRATCH / "p5t3-producer-planning2-original-test.py",
    SCRATCH / "p5t3-producer-pure2-coverage.json",
]
raw_paths.extend(sorted(SCRATCH.glob("p5t3-producer-pure*.stdout")))

raw_paths.extend(sorted(SCRATCH.glob("p5t3-producer-planning*.stdout")))
raw_paths.extend(sorted(SCRATCH.glob("p5t3-producer-planning*.stderr")))
entries = []
for path in raw_paths:
    raw = path.read_bytes()
    encoded = base64.b64encode(gzip.compress(raw, mtime=0)).decode("ascii")
    assert gzip.decompress(base64.b64decode(encoded)) == raw
    entries.append({
        "identifier": "SCRATCH/" + path.name, "sha256": digest(raw), "bytes": len(raw),
        "encoding": "gzip+base64", "data": encoded, "roundtrip_verified": True,
    })
for path in SCOPE:

    encoded = base64.b64encode(gzip.compress(raw, mtime=0)).decode("ascii")

        "identifier": path, "sha256": digest(raw), "bytes": len(raw),
        "encoding": "gzip+base64", "data": encoded, "roundtrip_verified": True,
    })

archive_path.write_bytes((json.dumps({"Timestamp": STAMP, "entries": entries}, indent=2, sort_keys=True) + "\n").encode("utf-8"))
