REG = FEATURE / 'evidence/regression-testing'
QA = FEATURE / 'evidence/qa-gates'


def sha(content):
    return hashlib.sha256(content).hexdigest()


def binding(path, expected=None):
    raw = (ROOT / path).read_bytes()
    digest = sha(raw)
    if expected is not None:
        assert digest == expected, str(path)
    return {'path': str(path).replace('\\', '/'), 'sha256': digest, 'bytes': len(raw)}


def read(path, expected=None):
    result = binding(path, expected)
    return json.loads((ROOT / path).read_bytes()), result


freeze, freeze_binding = read(REG / 'p4t7-integration-freeze.2026-10-19T10-53.json', '28975f0cd0dd00c11c1595135e9832ae502a8039c1c0d990c1b5ae034e5e4266')
sources = {item['path']: item['sha256'] for item in freeze['source_files']}
source_bindings = []
for path, digest in sources.items():
    item = binding(path, digest)
    raw = (ROOT / path).read_bytes()
    item['physical_lines'] = len(raw.splitlines())
    assert item['physical_lines'] <= 500 and b'\r' not in raw
    nodes = [node for node in ast.walk(ast.parse(raw)) if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef))]
    assert all(ast.get_docstring(node) for node in nodes)
    source_bindings.append(item)
borrowed = [binding(path, value['sha256']) for path, value in freeze['borrowed_sources'].items()]
assert len(source_bindings) == 3 and len(borrowed) == 5

assert freeze['test_count'] == 66 and len(freeze['test_nodes_in_execution_order']) == 66
