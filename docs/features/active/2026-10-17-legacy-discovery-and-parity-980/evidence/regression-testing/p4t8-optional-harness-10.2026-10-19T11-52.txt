def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def reference(path):
    return {'path': path.as_posix(), 'sha256': sha(path.read_bytes())}

def write(path, raw):
    assert not path.exists(), path
    path.write_bytes(raw)

def save(path, value):
    write(path, (json.dumps(value, indent=2) + '\n').encode())

    lines = text.splitlines()
    tree = ast.parse(text)
    definitions = [node for node in ast.walk(tree) if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef))]
    loops = [node for node in ast.walk(tree) if isinstance(node, (ast.For, ast.While))]
    assert len(lines) <= 500, (name, len(lines))
    assert all(ast.get_docstring(node) for node in definitions), name
    assert all(lines[node.lineno - 2].strip().startswith('#') for node in loops), name
    assert re.search(r'\b(?:tmp_path|tmpdir|tempfile|subprocess|requests|socket)\b', text) is None, name
    sources.append({'path': name, 'sha256': sha(raw), 'physical_lines': len(lines),

        'git_attributes': subprocess.check_output(['git', 'check-attr', '--all', '--', name], text=True).splitlines(),
        'definitions_with_docstrings': len(definitions), 'loops_with_immediate_intent_comments': len(loops)})

def assignments(raw):
    values = {}
    for node in ast.parse(raw).body:
        targets = node.targets if isinstance(node, ast.Assign) else [node.target] if isinstance(node, ast.AnnAssign) else []
        for target in targets:
            if isinstance(target, ast.Name):
                values[target.id] = ast.dump(node.value, include_attributes=False)
    return values
