def load(path):
    return json.loads(Path(path).read_text(encoding='utf-8-sig'))


def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def ref(path):
    return {'path': Path(path).as_posix(), 'sha256': digest(path)}

class CountedTokens(tuple):
    def __new__(cls, values):
        value = super().__new__(cls, values)
        value.accesses = 0
        return value

    def __getitem__(self, key):
        if isinstance(key, int):
            self.accesses += 1
        return super().__getitem__(key)


work_results = []
for case in prior['bounded_work_cases']:
    size = case['repetitions']
    source = 'class T<' * size + 'int' + '>' * size if case['shape'] == 'generic' else 'event Action ' * size + ';'
    assert hashlib.sha256(source.encode()).hexdigest() == case['input_sha256']
    tokens = CountedTokens(tokenize_csharp(source))
    error_code = None
    returned = False
    with patch.object(declarations, 'tokenize_csharp', return_value=tokens):
        try:
            declarations.declaration_candidates('src/work.cs', source, include_members=False)
            returned = True
        except SourceScanError as error:
            error_code = error.code
    assert error_code == 'source-work-limit' and not returned
    work_results.append({'shape': case['shape'], 'repetitions': size, 'input_sha256': case['input_sha256'], 'tokens': len(tokens), 'before_token_index_accesses': case['token_index_accesses'], 'after_token_index_accesses': tokens.accesses, 'error_code': error_code, 'partial_result_returned': returned})
