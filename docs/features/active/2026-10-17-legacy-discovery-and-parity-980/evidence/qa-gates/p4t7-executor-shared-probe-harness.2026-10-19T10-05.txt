"""Bound independent shared-contract reproductions without running worker suites."""

import hashlib
import json
import multiprocessing
import time
from datetime import datetime, timezone
from pathlib import Path

from drm_copilot_discovery.analyzer.dependency_inputs import (
    DependencyAnchor,
    DependencyBudget,
    DependencyInputError,
    DependencyLimits,
    dependency_value,
)


def version_child(connection, repetitions):
    """Classify one short memory value; send measured cost without a timing assertion."""
    value = '1-' + 'a-' * repetitions + '!'
    started = time.perf_counter()
    result = dependency_value(value)
    connection.send({'seconds': time.perf_counter() - started, 'kind': result.kind})
    connection.close()


class ObservedText(str):
    """Expose an actual UTF-8 allocation boundary to the isolated public collector."""

    def encode(self, encoding='utf-8', errors='strict'):
        """Record the allocated byte size; preserve normal str encoding semantics."""
        encoded = super().encode(encoding, errors)
        self.observations.append({'characters': len(self), 'allocated_bytes': len(encoded)})
        return encoded


def run():
    """Preserve raw probe results and exact executed bytes under canonical evidence."""
    root = Path.cwd()
    feature = Path('docs/features/active/2026-10-17-legacy-discovery-and-parity-980')
    qa = root / feature / 'evidence/qa-gates'
    paths = {
        'packages/discovery-runtime/src/drm_copilot_discovery/analyzer/dependency_inputs.py': '3b55bb3f4c79348851d2e8838c7233ab26aee1274526f916b55e2443d6289253',
        'tests/discovery_runtime/analyzer/test_dependency_inputs.py': '474422ef600bf8e68c8804cd0193aea6fe3b2ecaef8e64818d7e3548fb4ba39a',
    }
    for path, expected in paths.items():
        assert hashlib.sha256((root / path).read_bytes()).hexdigest() == expected
    measurements = []
    context = multiprocessing.get_context('spawn')
    # Each pure classification has an independent hard operational timeout.
    for repetitions in (8, 12, 16, 20, 28):
        parent, child = context.Pipe(duplex=False)
        process = context.Process(target=version_child, args=(child, repetitions))
        started = time.perf_counter()
        process.start()
        child.close()
        process.join(timeout=2.0)
        if process.is_alive():
            process.terminate()
            process.join(timeout=2.0)
            observation = {'status': 'operational-timeout', 'child_budget_seconds': 2.0}
        else:
            assert process.exitcode == 0 and parent.poll()
            observation = dict(parent.recv(), status='returned')
        parent.close()
        measurements.append(dict(observation, repetitions=repetitions,
                                 input_characters=3 + 2 * repetitions,
                                 parent_elapsed_seconds=time.perf_counter() - started))

    value = ObservedText('\U0001f600' * 3)
    value.observations = []
    budget = DependencyBudget(DependencyLimits(max_text_bytes=5))
    try:
        budget.add(kind='declaration', ecosystem='python', name=value, version=None,
                   role='install', anchor=DependencyAnchor('text', 'requirements', line=1))
    except DependencyInputError as error:
        allocation = {'configured_text_capacity': 5, 'observations': value.observations,
                      'result_code': error.code}
    else:
        raise AssertionError('Expected limit rejection')
    try:
        budget.result()
    except DependencyInputError:
        allocation['partial_result_refused'] = True

    classifications = []
    for text in ('1.2.3', '*', '1.*', '1.x', '1.2.*', '1.X'):
        result = dependency_value(text)
        classifications.append({'input': text, 'kind': result.kind, 'value': result.value})
    for path, expected in paths.items():
        assert hashlib.sha256((root / path).read_bytes()).hexdigest() == expected
    script = Path(__file__).read_bytes()
    stamp = datetime.now(timezone.utc).strftime('%Y-%m-%dT%H-%M')
    harness = qa / f'p4t7-executor-shared-probe-harness.{stamp}.txt'
    with harness.open('xb') as stream:
        stream.write(script)
    now = datetime.now(timezone.utc)
    stamp = now.strftime('%Y-%m-%dT%H-%M')
    receipt = qa / f'p4t7-executor-shared-probe.{stamp}.json'
    payload = {
        'timestamp': stamp, 'published_utc': now.isoformat(), 'exit_code': 0,
        'command': 'poetry run python SCRATCH/issue-980-p4t7-shared-probe.py',
        'scope': 'Immutable dependency_inputs shared subset only; no worker or predecessor suite.',
        'source_bindings': paths, 'regex_input_recipe': "'1-' + 'a-' * repetitions + '!'",
        'regex_observations': measurements,
        'timing_qualification': 'Diagnostic measurements with operational termination only; no unit-test timing threshold or whole-task acceptance claim.',
        'prospective_encoding': allocation, 'version_classifications': classifications,
        'exact_executed_harness': {'path': harness.relative_to(root).as_posix(),
                                   'sha256': hashlib.sha256(script).hexdigest()},
    }
    with receipt.open('xb') as stream:
        stream.write((json.dumps(payload, indent=2) + '\n').encode('utf-8'))
    print(json.dumps({'receipt': receipt.relative_to(root).as_posix(),
                      'sha256': hashlib.sha256(receipt.read_bytes()).hexdigest(),
                      'observations': payload}, indent=2))


if __name__ == '__main__':
    run()
