def sha(data):
    """Bind exact original bytes without encoding or newline conversion."""
    return hashlib.sha256(data).hexdigest()


def reference(path):
    """Return a portable canonical artifact reference with exact byte identity."""
    raw = path.read_bytes()
    return {"path": path.relative_to(ROOT).as_posix(), "sha256": sha(raw), "bytes": len(raw)}


def write(path, value):
    """Write one new LF UTF-8 preparation receipt without replacing any predecessor."""
    if path.exists():
        raise ValueError("Preparation evidence already exists")
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n",
                    encoding="utf-8", newline="\n")
    return reference(path)


def variants(path):
    """Observe Windows short/long spellings for readable redaction; perform no writes."""
    values = {str(path), path.as_posix()}
    if os.name == "nt":
        # Both actual Windows aliases must map to the same portable evidence token.
        for name in ("GetLongPathNameW", "GetShortPathNameW"):
            buffer = ctypes.create_unicode_buffer(32768)
            size = getattr(ctypes.windll.kernel32, name)(str(path), buffer, len(buffer))
            if not 0 < size < len(buffer):
                raise ValueError("Cannot observe the native path alias")
            values.update((buffer.value, buffer.value.replace("\\", "/")))
    return values
