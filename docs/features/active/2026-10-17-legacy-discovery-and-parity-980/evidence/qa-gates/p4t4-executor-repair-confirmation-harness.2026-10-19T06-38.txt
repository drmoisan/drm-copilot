def sha(raw: bytes) -> str:
    """Return SHA-256 without changing immutable bytes."""
    return hashlib.sha256(raw).hexdigest()


def portable(value: str) -> str:
    """Reproduce only the retained worker's explicit host-path normalization."""
    for path, label in ((ROOT, "WORKSPACE"), (WORKER, "SCRATCH")):
        value = re.sub(re.escape(str(path)), label, value, flags=re.IGNORECASE)
        value = re.sub(re.escape(path.as_posix()), label, value, flags=re.IGNORECASE)
    return value


def executable_definition(node: ast.FunctionDef | ast.AsyncFunctionDef | ast.ClassDef) -> str:
    """Return AST including decorators/assertions and excluding a leading docstring."""
    if node.body and isinstance(node.body[0], ast.Expr) and isinstance(node.body[0].value, ast.Constant) and isinstance(node.body[0].value.value, str):
        node.body = node.body[1:]
    return ast.dump(node, include_attributes=False)

class CountedMatcher:
    """Count complete linear-match input for the two actual INI parsing stages."""
    def match(self, text: str):
        """Return the actual regex result after recording str input length locally."""
        visits.append(len(text))
        return linear.match(text)


class ProhibitedMatcher:
    """Reject accidental fallback to the original overlapping stdlib expression."""
    def match(self, text: str):
        """Raise before using the str input; no source values are exposed."""
        raise AssertionError("Inherited overlapping option expression invoked")
