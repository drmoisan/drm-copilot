"""Bind completed scoped P4-T9 proof without rerunning behavior or changing code."""
import ast
import base64
import copy
import datetime
import gzip
import hashlib
import json
from pathlib import Path
import re

repo, scratch = Path.cwd(), Path(__file__).parent
feature = repo / 'docs/features/active/2026-10-17-legacy-discovery-and-parity-980'
evidence = feature / 'evidence/regression-testing'
stamp = datetime.datetime.now(datetime.timezone.utc).strftime('%Y-%m-%dT%H-%M')
owned = [
    'packages/discovery-runtime/src/drm_copilot_discovery/extension_results.py',
    'packages/discovery-runtime/src/drm_copilot_discovery/extension_validation.py',
    'tests/discovery_runtime/test_extension_results.py',
    'tests/discovery_runtime/test_extension_inventory_merge.py',
    'tests/discovery_runtime/test_extension_validation_limits.py',
]
label = 'p4t9-core-repair3'


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def binding(path):
    raw = path.read_bytes()
    return {'path': path.relative_to(repo).as_posix(), 'sha256': sha(raw), 'bytes': len(raw)}


def write_json(name, value):
    path = evidence / f'{name}.{stamp}.json'
    assert not path.exists(), str(path)
    path.write_bytes((json.dumps(value, indent=2) + '\n').encode())
    return path


receipts = [next(evidence.glob(f'{label}-{stage}.*.json'))
            for stage in ('black', 'ruff', 'pyright', 'precheck', 'pytest')]
records = [json.loads(path.read_bytes()) for path in receipts]
assert all(value['exit_code'] == 0 for value in records)
source = {path: {'sha256': sha((repo / path).read_bytes()),
                 'physical_lines': len((repo / path).read_bytes().splitlines())}
          for path in owned}
for value in records:
    assert value['source_before'] == value['source_after'] == source
    assert value['borrowed_before'] == value['borrowed_after']
    assert value['borrowed_before'] == records[-1]['borrowed_after']
stdout = records[-1]['streams']['stdout']['presentation']
nodes = re.findall(r'^(tests/.+?) (?:PASSED)(?:\s|$)', stdout, re.M)
assert len(nodes) == len(set(nodes)) == 144
assert '144 passed' in stdout and 'FAILED' not in stdout
coverage_path = scratch / f'{label}-coverage.json'
coverage = json.loads(coverage_path.read_bytes())
coverage_files = {name.replace('\\', '/'): value for name, value in coverage['files'].items()}
summaries, findings = [], []
for path in owned:
    raw = (repo / path).read_bytes()
    lines = raw.decode('utf-8').splitlines()
    tree = ast.parse(raw)
    if len(lines) > 500:
        findings.append([path, 'physical-line-count', len(lines)])
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            if not ast.get_docstring(node):
                findings.append([path, 'missing-docstring', node.lineno])
        if isinstance(node, (ast.For, ast.While)):
            preceding = next((line.strip() for line in reversed(lines[:node.lineno - 1])
                              if line.strip()), '')
            if not preceding.startswith('#'):
                findings.append([path, 'missing-loop-intent', node.lineno])
    if path not in coverage_files:
        continue
    data = coverage_files[path]
    ranges = []
    for family in ('classes', 'functions'):
        for name, value in data[family].items():
            if not name:
                continue
            summary = value['summary']
            if summary['num_statements']:
                assert summary['covered_lines'] / summary['num_statements'] >= .9, (path, name)
            ranges.append({'kind': family, 'name': name, 'summary': summary,
                           'executed_lines': value['executed_lines'],
                           'missing_lines': value['missing_lines']})
    assert data['summary']['covered_lines'] / data['summary']['num_statements'] >= .9
    summaries.append({'path': path, 'summary': data['summary'], 'ranges': ranges,
                      'executed_lines': data['executed_lines'],
                      'missing_lines': data['missing_lines'],
                      'missing_branches': data['missing_branches']})
assert not findings, findings

# Reconcile relocated bodies/decorators and every earlier case without replay.
original = scratch / 'p4t9-core-before-repair-2026-10-19T13-37'
def definitions(paths):
    return {node.name: (path, node) for path in paths
            for node in ast.parse(path.read_bytes()).body if isinstance(node, ast.FunctionDef)}


def normalized(node):
    result = copy.deepcopy(node)
    for child in ast.walk(result):
        if isinstance(child, (ast.FunctionDef, ast.ClassDef)) and ast.get_docstring(child):
            child.body = child.body[1:]
    return ast.dump(result, include_attributes=False)


old_definitions = definitions([original / Path(path).name for path in owned[2:4]])
new_definitions = definitions([repo / path for path in owned[2:]])
assert set(old_definitions) <= set(new_definitions)
changed = [name for name, (_, node) in old_definitions.items()
           if normalized(node) != normalized(new_definitions[name][1])]
assert changed == ['test_capture_index_count_is_bounded_before_projection'], changed
old_receipt = json.loads(next(evidence.glob('p4t9-core-focused11-pytest.*.json')).read_bytes())
old_nodes = re.findall(r'^(tests/.+?) (?:PASSED|FAILED)(?:\s|$)', old_receipt['streams']['stdout']['presentation'], re.M)
assert len(old_nodes) == 125
old_cases = {node.split('::', 1)[1] for node in old_nodes}
new_cases = {node.split('::', 1)[1] for node in nodes}
assert old_cases <= new_cases and len(new_cases - old_cases) == 19
moved = [{'name': name, 'from': old_path.name, 'to': new_definitions[name][0].name,
          'behavior_ast_unchanged': name not in changed}
         for name, (old_path, _) in old_definitions.items()
         if old_path.name != new_definitions[name][0].name]
assert len(moved) == 7
relocation = write_json('p4t9-core-repair-preservation', {
    'review': 'evidence/qa-gates/p4t9-executor-shared-core-review.2026-10-19T13-31.json',
    'original_test_paths': {path: sha((original / Path(path).name).read_bytes()) for path in owned[2:4]},
    'retained_definitions': len(old_definitions), 'moved_definitions': moved,
    'only_original_behavior_delta': {'name': changed[0], 'change': 'Use explicit low profile limits instead of reading an omitted optional limits member.'},
    'all_125_original_case_ids_retained': True, 'new_case_ids': sorted(new_cases - old_cases),
    'docstrings_ignored_for_body_comparison': True,
})

# Exact encoded originals remain recoverable without exposing host-path text in views.
archive_entries = []
raw_paths = sorted(set(scratch.glob('p4t9-core-*.bin'))
                   | set(scratch.glob('p4t9-core-*-coverage.json'))
                   | set(scratch.glob('p4t9-core-*.coverage'))
                   | set(scratch.glob('p4t9-core-*.py'))
                   | set(scratch.glob('p4t9-core-moved-case-inventory.json'))
                   | set(original.iterdir()))
for path in raw_paths:
    raw = path.read_bytes()
    encoded = base64.b64encode(gzip.compress(raw, mtime=0)).decode('ascii')
    assert gzip.decompress(base64.b64decode(encoded)) == raw
    archive_entries.append({'scratch_identifier': 'SCRATCH/' + path.relative_to(scratch).as_posix(),
                            'sha256': sha(raw), 'bytes': len(raw),
                            'encoding': 'base64(gzip(raw)); gzip mtime=0', 'data': encoded})
archive = write_json('p4t9-core-original-streams', {'decode_verified': True, 'entries': archive_entries})
views = []
for name in ('p4t9-core-gate.py', 'p4t9-core-coverage.py', 'p4t9-core-wrap-patch.py',
             'p4t9-core-freeze.py', 'p4t9-core-repair-baseline.py', 'p4t9-core-repair-tests.py'):
    raw = (scratch / name).read_bytes()
    text = raw.decode('utf-8').replace('\r\n', '\n').replace('\r', '\n')
    text = '\n'.join(line.rstrip() for line in text.splitlines()).rstrip() + '\n'
    path = evidence / f'{Path(name).stem}-harness.{stamp}.txt'
    assert not path.exists()
    path.write_bytes(text.encode())
    views.append({'original_sha256': sha(raw), 'original_archive_entry': 'SCRATCH/' + name,
                  'readable': binding(path), 'transform': 'LF line endings; trailing whitespace/terminal blank lines trimmed only'})
proof = write_json('p4t9-core-coverage-structure', {
    'source': source, 'original_coverage_sha256': sha(coverage_path.read_bytes()),
    'original_coverage_archive_entry': 'SCRATCH/' + coverage_path.name,
    'summaries': summaries, 'structure_findings': findings,
    'new_and_changed_lines': 'All owned production paths are new relative to accepted baseline; every executable line remains in measurement.',
})

    'accepted_baseline': 'c64b35435d1c1597830d8dd03e661d7944192065',
    'source': source, 'borrowed': records[-1]['borrowed_after'],
    'ordered_receipts': [binding(path) for path in receipts],
    'cases': len(nodes), 'forward_nodes': nodes, 'unit_seconds': records[-1]['duration_seconds'],
    'coverage_structure': binding(proof), 'exact_original_archive': binding(archive),
    'repair_preservation': binding(relocation), 'precheck_cases': 20,
    'harness_views': views,
    'baseline_reference': 'evidence/baseline/p4t9-shared-baseline.2026-10-19T12-41.json',
    'helper_absence_reference': 'evidence/baseline/p4t9-core-helper-absence.2026-10-19T12-50.json',
    'qualifications': [
        'Original focused6 retained 111 passes and one invalid fixture-category failure; focused10 retained 123 passes and two invalid fixture constructions; focused11 retained 124 passes and one optional-limits fixture failure. All diagnostics and raw originals remain recoverable.',
        'The complete R2/R3 repair preserves 125 original cases, adds 19 counted integer boundaries, moves seven test definitions and completes required docstrings. Only the original count-limit fixture body changes.',
        'Integer byte reservations use bounded power comparisons without decimal rendering; signs, separator and existing usage count before allocation. The 20-case precheck precedes the 144-case affected population.',
        'Readable command streams sanitize only workspace/SCRATCH paths; Markdown also normalizes whitespace. Exact originals decode from the archive with verified hashes.',
        'The helper absence receipt was renamed to its recorded UTC minute after a suffix-handling error; exact original bytes and executed scope harness are retained.',
        'Shared dependency windows changed only between retained runs; this clean sequence binds one common exact window.',
        'Original console-only import-sorting preparation and a failed read-only default-encoding preview are not asserted as quality gates.',
        'Documentation is included in input digest/accounting, but current loader source claims support source/test/entrypoints only.',
        'No data-validation result authorizes publication; actual capture/loading composition and full final ordered gates remain pending.',

freeze = write_json('p4t9-core-freeze', manifest)
md = evidence / f'p4t9-core-freeze.{stamp}.md'
md.write_bytes((f'# P4-T9 core checked subset\n\nTimestamp: {stamp}\n\n'

stage = sorted(set(owned) | {path.relative_to(repo).as_posix() for path in evidence.glob('p4t9-core-*')}
