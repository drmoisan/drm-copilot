sha = lambda raw: hashlib.sha256(raw).hexdigest()


def ref(path):
    return {"path": str(path).replace("\\", "/"), "sha256": sha(Path(path).read_bytes())}


def bound(reference):
    raw = Path(reference["path"]).read_bytes()
    assert sha(raw) == reference["sha256"], reference["path"]
    if "bytes" in reference:
        assert len(raw) == reference["bytes"]
    result = json.loads(raw)
    for key in reference.get("selector", []):
        result = result[key]
    return result


def fixed(name, digest):
    return {"path": (Q / name).as_posix(), "sha256": digest}

def decode(reference):
    result = {}
    for row in bound(reference)["entries"]:
        raw = base64.b64decode(row["base64"], validate=True)
        assert len(raw) == row["bytes"] and sha(raw) == row["sha256"], row["path"]
        assert row["path"] not in result
        result[row["path"]] = raw
    return result

aliases = []
for root, label in roots:
    spellings = {str(root), root.as_posix()}
    for name in ("GetLongPathNameW", "GetShortPathNameW"):
        buffer = ctypes.create_unicode_buffer(32768)
        size = getattr(ctypes.windll.kernel32, name)(str(root), buffer, len(buffer))
        assert 0 < size < len(buffer) and root.samefile(buffer.value)
        spellings.update((buffer.value, buffer.value.replace("\\", "/")))
    aliases.extend((spelling, label) for spelling in spellings)


def readable(raw):
    text = raw.decode("utf-8")
    for spelling, label in sorted(aliases, key=lambda x: len(x[0]), reverse=True):
        text = re.sub(re.escape(spelling), lambda match: label, text, flags=re.I)
    return text.replace("\r\n", "\n").replace("\r", "\n")
