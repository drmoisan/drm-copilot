"""Execute one reviewed P4-T2 ordered gate loop with immutable candidate bindings."""
import datetime
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys

root=Path.cwd()
scratch=Path(__file__).resolve().parent
qa=Path('docs/features/active/2026-10-17-legacy-discovery-and-parity-980/evidence/qa-gates')
manifest_path=Path(sys.argv[1])
manifest=json.loads(manifest_path.read_text(encoding='utf-8'))
expected={entry['path']:entry['working_sha256'] for entry in manifest['files']}
code=[path for path in expected if path.endswith('.py')]
receipts=[]
harnesses=[Path(__file__).resolve(),scratch/'architecture_gate.py',scratch/'native_gate.py']
harness_hashes={'SCRATCH/'+path.name:hashlib.sha256(path.read_bytes()).hexdigest() for path in harnesses}
start=datetime.datetime.now(datetime.timezone.utc)
def hashes():
    return {path:hashlib.sha256(Path(path).read_bytes()).hexdigest() for path in expected}
def portable(text):
    for path,label in ((root,'WORKSPACE'),(scratch,'SCRATCH')):
        text=re.sub(re.escape(str(path)),label,text,flags=re.IGNORECASE)
        text=re.sub(re.escape(path.as_posix()),label,text,flags=re.IGNORECASE)
    return text
def run(number,label,arguments,*,module=True):
    before=hashes()
    assert before==expected,('candidate changed before',label)
    started=datetime.datetime.now(datetime.timezone.utc)
    env=dict(os.environ)
    env.update(PYTHONIOENCODING='utf-8',PYTHONDONTWRITEBYTECODE='1',COVERAGE_FILE=str(scratch/'p4t2-final.coverage'))
    command=[sys.executable,*(['-m'] if module else []),*map(str,arguments)]
    result=subprocess.run(command,capture_output=True,env=env)
    finished=datetime.datetime.now(datetime.timezone.utc)
    stamp=finished.strftime('%Y-%m-%dT%H-%M')
    stem='p4t2-final-'+label+'-'+finished.strftime('%H%M%S')+'.'+stamp
    stdout=scratch/(stem+'.stdout.txt')
    stderr=scratch/(stem+'.stderr.txt')
    stdout.write_bytes(result.stdout)
    stderr.write_bytes(result.stderr)
    after=hashes()
    document={'stage':number,'label':label,'timestamp':stamp,'started_at':started.isoformat(),'finished_at':finished.isoformat(),'seconds':(finished-started).total_seconds(),'command':portable('poetry run python '+('-m ' if module else '')+subprocess.list2cmdline(list(map(str,arguments)))),'exit_code':result.returncode,'source_before':before,'source_after':after,'stdout':portable(result.stdout.decode('utf-8',errors='replace')),'stderr':portable(result.stderr.decode('utf-8',errors='replace')),'raw_stdout':{'path':'SCRATCH/'+stdout.name,'sha256':hashlib.sha256(result.stdout).hexdigest()},'raw_stderr':{'path':'SCRATCH/'+stderr.name,'sha256':hashlib.sha256(result.stderr).hexdigest()},'harness_hashes':harness_hashes}
    path=qa/(stem+'.json')
    path.write_text(json.dumps(document,indent=2,sort_keys=True)+'\n',encoding='utf-8',newline='\n')
    markdown=f'# P4-T2 final {label}\n\nTimestamp: {stamp}\nCommand: `{document["command"]}`\nEXIT_CODE: {result.returncode}\nOutput Summary: Stage {number} of7; exact source bindings, timestamps and raw output hashes are in the adjacent JSON.\n\n```text\n{document["stdout"]}{document["stderr"]}\n```\n'
    path.with_suffix('.md').write_text('\n'.join(line.rstrip() for line in markdown.splitlines())+'\n',encoding='utf-8',newline='\n')
    receipts.append({'stage':number,'label':label,'path':path.as_posix(),'sha256':hashlib.sha256(path.read_bytes()).hexdigest(),'exit_code':result.returncode})
    progress=qa/('p4t2-final-progress-'+start.strftime('%H%M%S')+'.'+stamp+'.json')
    progress.write_text(json.dumps({'candidate_manifest':manifest_path.as_posix(),'receipts':receipts,'source_hashes':after},indent=2,sort_keys=True)+'\n',encoding='utf-8',newline='\n')
    print(json.dumps({'stage':number,'label':label,'exit_code':result.returncode,'seconds':document['seconds'],'receipt':path.as_posix()}),flush=True)
    if result.returncode or before!=after:
        print((document['stdout'][-8000:]+document['stderr'][-8000:]).encode('ascii',errors='backslashreplace').decode('ascii'),flush=True)
        raise SystemExit(result.returncode or 1)

assert hashes()==expected
for path in expected:
    subprocess.run(['git','ls-files','--error-unmatch','--',path],capture_output=True,check=True)
run(1,'format',['black','--check',*code])
run(2,'lint',['ruff','check',*code])
run(3,'types',['pyright'])
run(4,'architecture',[scratch/'architecture_gate.py',manifest_path],module=False)
coverage_raw=scratch/'p4t2-final-coverage.raw.json'
tests=['tests/discovery_runtime/analyzer/test_dotnet_projects.py','tests/discovery_runtime/analyzer/test_solution_input.py','tests/discovery_runtime/analyzer/test_xml_input.py']
run(5,'tests-reverse',['pytest','-q','-o','addopts=',*tests,'--cov=drm_copilot_discovery.analyzer.xml_input','--cov=drm_copilot_discovery.analyzer.solution_input','--cov=drm_copilot_discovery.analyzer.dotnet_projects','--cov-branch','--cov-report=term-missing','--cov-report=json:'+coverage_raw.as_posix()])
stamp=datetime.datetime.now(datetime.timezone.utc).strftime('%Y-%m-%dT%H-%M')
coverage_path=qa/f'p4t2-final-coverage.{stamp}.json'
coverage_path.write_bytes(coverage_raw.read_bytes())
coverage=json.loads(coverage_path.read_text())
for filename,data in coverage['files'].items():
    assert data['summary']['percent_statements_covered']>=90,filename
    assert data['summary']['percent_branches_covered']>=90,filename
    for family in ('functions','classes'):
        for name,unit in data.get(family,{}).items():
            summary=unit['summary']
            if summary['num_statements']:
                assert summary['percent_statements_covered']>=90,(filename,name,summary)
run(6,'contracts',['pytest','-q','-o','addopts=','--no-cov','tests/schemas/discovery/test_v2_inventory_schemas.py','tests/scripts/dev_tools/test_quality_tiers_contract.py','tests/scripts/dev_tools/test_quality_tiers_contract_classification.py','tests/discovery_runtime/test_package_contract.py::test_runtime_metadata_has_bounded_dependencies_and_minimum_python','tests/discovery_runtime/test_package_contract.py::test_root_tooling_includes_the_local_runtime_for_types_and_coverage'])
native_raw=scratch/'p4t2-final-native-result.raw.json'
run(7,'native',[scratch/'native_gate.py',native_raw],module=False)
native_path=qa/('p4t2-final-native-result.'+datetime.datetime.now(datetime.timezone.utc).strftime('%Y-%m-%dT%H-%M')+'.json')
native_path.write_bytes(native_raw.read_bytes())
finished=datetime.datetime.now(datetime.timezone.utc)
path=qa/('p4t2-final-ordered-stages.'+finished.strftime('%Y-%m-%dT%H-%M')+'.json')
path.write_text(json.dumps({'task':'P4-T2','candidate_manifest':manifest_path.as_posix(),'candidate_manifest_sha256':hashlib.sha256(manifest_path.read_bytes()).hexdigest(),'started_at':start.isoformat(),'finished_at':finished.isoformat(),'receipts':receipts,'source_hashes':hashes(),'harness_hashes':harness_hashes,'test_module_order':tests,'coverage_path':coverage_path.as_posix(),'native_path':native_path.as_posix(),'raw_coverage':{'path':'SCRATCH/'+coverage_raw.name,'sha256':hashlib.sha256(coverage_raw.read_bytes()).hexdigest()},'raw_native':{'path':'SCRATCH/'+native_raw.name,'sha256':hashlib.sha256(native_raw.read_bytes()).hexdigest()},'host_proof_limitations':'Native Windows Python/Git only; coding-agent host execution and non-Windows CI are not exercised by this task.'},indent=2,sort_keys=True)+'\n',encoding='utf-8',newline='\n')
print(path.as_posix(),flush=True)
