def sha(content: bytes) -> str:
    """Return the SHA-256 identity of supplied bytes without external effects."""
    return hashlib.sha256(content).hexdigest()


def reference(path: Path) -> dict[str, object]:
    """Read one existing artifact and return its relative identity and size."""
    content = path.read_bytes()
    return {"path": path.as_posix(), "sha256": sha(content), "bytes": len(content)}


def save(path: Path, content: bytes) -> None:
    """Create one LF-only preparation artifact without replacing any prior file."""
    assert not path.exists(), path
    assert b"\r" not in content
    assert content.endswith(b"\n") and not content.endswith(b"\n\n")
    assert all(line.rstrip() == line for line in content.splitlines())
    path.write_bytes(content)
