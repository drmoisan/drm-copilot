def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def reference(path):
    return {"path": path.as_posix(), "sha256": sha(path.read_bytes())}


def write(path, raw):
    assert not path.exists(), path.name
    path.write_bytes(raw)


def json_write(path, value):
    write(path, (json.dumps(value, indent=2) + "\n").encode())


class WithoutDocs(ast.NodeTransformer):
    def visit_Constant(self, node):
        return node

    def generic_visit(self, node):
        node = super().generic_visit(node)
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)) and ast.get_docstring(node):
            node.body = node.body[1:]
        return node

sources = []
for name in paths:
    raw = Path(name).read_bytes()
    text = raw.decode()
    lines = text.splitlines()
    tree = ast.parse(text)
    definitions = [node for node in ast.walk(tree) if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef))]
    loops = [node for node in ast.walk(tree) if isinstance(node, (ast.For, ast.While))]
    assert len(lines) <= 500 and all(ast.get_docstring(node) for node in definitions)
    assert all(lines[node.lineno - 2].strip().startswith("#") for node in loops)
    assert re.search(r"\b(?:tmp_path|tmpdir|tempfile|subprocess|requests|socket)\b", text) is None
    sources.append({"path": name, "sha256": sha(raw), "physical_lines": len(lines),
                    "lf_normalized_sha256": sha(raw.replace(b"\r\n", b"\n")), "crlf_count": raw.count(b"\r\n"),
                    "git_attributes": subprocess.check_output(["git", "check-attr", "--all", "--", name], text=True).splitlines(),
                    "definitions_with_docstrings": len(definitions), "loops_with_immediate_intent_comments": len(loops)})
