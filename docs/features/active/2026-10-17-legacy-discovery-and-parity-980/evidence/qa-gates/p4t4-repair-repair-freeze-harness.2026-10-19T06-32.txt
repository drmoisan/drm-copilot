"""Bind the four-file P4-T4 repair and retained tests without task acceptance."""
import ast
from datetime import datetime, timezone
from hashlib import sha256
import json
from pathlib import Path
import subprocess

root = Path.cwd()
scratch = Path(__file__).resolve().parent
evidence = Path('docs/features/active/2026-10-17-legacy-discovery-and-parity-980/evidence')
qa = evidence / 'qa-gates'
binding_path = qa / 'p4t4-repair-7-binding.2026-10-19T06-28.json'
original_path = qa / 'p4t4-focused-wip-freeze.2026-10-19T05-54.json'
review_path = qa / 'p4t4-executor-candidate-review.2026-10-19T06-10.json'
baseline_path = evidence / 'baseline/p4t4-baseline-binding.2026-10-19T05-38.json'
binding = json.loads(binding_path.read_bytes())
original = json.loads(original_path.read_bytes())
coverage = json.loads(Path(binding['coverage']['path']).read_bytes())
old_coverage = json.loads(Path(original['coverage']['path']).read_bytes())
assert sha256(review_path.read_bytes()).hexdigest() == 'eff5e903958412fc4fdf09b305481dd9788d905d8e2c4867969736a4d6b79990'

def ref(path):
    return {'path': path.as_posix(), 'sha256': sha256(path.read_bytes()).hexdigest()}

def executable_dump(node):
    copied = ast.parse(ast.unparse(node)).body[0]
    for child in ast.walk(copied):
        if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)) and ast.get_docstring(child):
            child.body.pop(0)
    return ast.dump(copied, include_attributes=False)

files = []
changed = []
original_functions = []
new_functions = []
units = []
module_coverage = []
documentation = []
old_files = {item['path']: item for item in original['files']}
for path, expected in binding['sources'].items():
    content = Path(path).read_bytes()
    actual = sha256(content).hexdigest()
    assert actual == expected, path
    old = old_files[path]
    preserved = scratch / 'original-reviewed' / path
    assert sha256(preserved.read_bytes()).hexdigest() == old['sha256'], path
    count = len(content.decode().splitlines())
    assert count <= 500, (path, count)
    disposition = 'repaired' if actual != old['sha256'] else 'unchanged'
    files.append({'path': path, 'sha256': actual, 'original_sha256': old['sha256'], 'physical_lines': count, 'disposition': disposition})
    if disposition == 'repaired':
        changed.append(path)
    if not path.endswith('.py'):
        continue
    tree = ast.parse(content)
    for node in ast.walk(tree):
        if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
            assert ast.get_docstring(node), (path, node.name)
            documentation.append({'path': path, 'name': node.name, 'line': node.lineno})
    if path.startswith('tests/'):
        old_nodes = {node.name: node for node in ast.parse(preserved.read_bytes()).body if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef))}
        new_nodes = {node.name: node for node in tree.body if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef))}
        for name, old_node in old_nodes.items():
            assert name in new_nodes and executable_dump(old_node) == executable_dump(new_nodes[name]), (path, name)
            original_functions.append({'path': path, 'name': name, 'assertions_and_parameters_preserved': True})
        new_functions.extend({'path': path, 'name': name} for name in new_nodes.keys() - old_nodes.keys())
    if path not in coverage['files']:
        continue
    data = coverage['files'][path]
    module_coverage.append({'path': path, 'original': old_coverage['files'][path]['summary'], 'repaired': data['summary'], 'missing_lines': data['missing_lines'], 'missing_branches': data['missing_branches']})
    assert data['summary']['percent_statements_covered'] >= old_coverage['files'][path]['summary']['percent_statements_covered']
    assert data['summary']['percent_branches_covered'] >= old_coverage['files'][path]['summary']['percent_branches_covered']
    eligible = set(data['executed_lines']) | set(data['missing_lines'])
    for node in ast.walk(tree):
        if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
            span = set(range(node.lineno, node.end_lineno + 1))
            statements = eligible & span
            covered = set(data['executed_lines']) & span
            assert not statements or len(covered) / len(statements) >= .90, (path, node.name)
            units.append({'path': path, 'name': node.name, 'kind': type(node).__name__, 'start': node.lineno, 'end': node.end_lineno, 'statements': len(statements), 'covered_statements': len(covered)})
assert len(changed) == 4
assert not any('fixtures/' in path for path in changed)

        assert sha256((scratch / Path(raw['path']).name).read_bytes()).hexdigest() == raw['sha256']
test_receipt = json.loads(Path(binding['receipts'][-1]['path']).read_bytes())
assert '91 passed' in test_receipt['stdout']
assert sha256((scratch / Path(binding['coverage']['raw_path']).name).read_bytes()).hexdigest() == binding['coverage']['raw_sha256']

    'original_test_definitions': original_functions, 'new_test_definitions': sorted(new_functions, key=lambda entry: (entry['path'], entry['name'])),

    'repair_history': history_refs, 'failed_run_coverage': {'path': 'SCRATCH/' + failed_coverage.name, 'sha256': sha256(failed_coverage.read_bytes()).hexdigest(), 'status': 'unaccepted fixture assertion failure; retained, not used for acceptance'},
