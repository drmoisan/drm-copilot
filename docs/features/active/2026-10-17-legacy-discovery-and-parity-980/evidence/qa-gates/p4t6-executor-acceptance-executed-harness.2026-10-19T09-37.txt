"""Publish the retained P4-T6 acceptance without replaying behavioral proof."""

import hashlib
import json
import re
import subprocess
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path.cwd()
FEATURE = Path('docs/features/active/2026-10-17-legacy-discovery-and-parity-980')
QA = FEATURE / 'evidence/qa-gates'
PLAN = FEATURE / 'plan.2026-10-17T22-38.md'
RECON = QA / 'p4t6-executor-final-reconciliation.2026-10-19T09-31.json'


def digest(data):
    """Return the byte identity used by the accepted evidence contract."""
    return hashlib.sha256(data).hexdigest()


def binding(path):
    """Record working and qualified LF publication identities separately."""
    raw = (ROOT / path).read_bytes()
    return {
        'path': path.as_posix(), 'working_sha256': digest(raw),
        'expected_lf_blob_sha256': digest(raw.replace(b'\r\n', b'\n')),
        'crlf_count': raw.count(b'\r\n'),
    }


def publish(stem, suffix, payload):
    """Use actual UTC publication time and avoid overwriting retained proof."""
    now = datetime.now(timezone.utc)
    minute = now.strftime('%Y-%m-%dT%H-%M')
    path = QA / f'{stem}.{minute}.{suffix}'
    if isinstance(payload, dict):
        payload = dict(payload, timestamp=minute, published_utc=now.isoformat())
        text = json.dumps(payload, indent=2, ensure_ascii=True) + '\n'
    else:
        text = payload.replace('{TIMESTAMP}', minute).replace('{WRITE_TIME}', now.isoformat())
    raw = text.encode('utf-8')
    assert b'\r' not in raw and raw.endswith(b'\n') and not raw.endswith(b'\n\n')
    assert all(line == line.rstrip() for line in raw.splitlines())
    with (ROOT / path).open('xb') as stream:
        stream.write(raw)
    return path
