def sha(data):
    """Return an exact byte digest."""
    return hashlib.sha256(data).hexdigest()


def reference(path):
    """Bind an existing repository-relative evidence stream."""
    return {"path": path.as_posix(), "sha256": sha(path.read_bytes())}


def read(path):
    """Load a retained JSON record without changing its bytes."""
    return json.loads(path.read_bytes())


def create(path, text):
    """Write only a new LF preparation stream, with one terminal newline."""
    assert not path.exists(), path
    value = text.replace("\r\n", "\n").rstrip() + "\n"
    assert "\r" not in value
    path.write_text(value, encoding="utf-8", newline="\n")
