def sha(data):
    """Identify exact bytes; no normalization is implicit."""
    return hashlib.sha256(data).hexdigest()


def bound(path, expected=None):
    """Read a retained artifact and fail if its frozen identity changed."""
    path = Path(path)
    data = (ROOT / path).read_bytes()
    identity = sha(data)
    if expected is not None:
        assert identity == expected, path.as_posix()
    return {'path': path.as_posix(), 'sha256': identity}
