sha = lambda raw: hashlib.sha256(raw).hexdigest()


def ref(path):
    return {'path': Path(path).as_posix(), 'sha256': sha(Path(path).read_bytes())}


def walk(value):
    if isinstance(value, dict):
        yield value
        for child in value.values():
            yield from walk(child)
    elif isinstance(value, list):
        for child in value:
            yield from walk(child)


def publish(stem, suffix, value):
    now = dt.datetime.now(dt.timezone.utc)
    path = Q / (stem + '.' + now.strftime('%Y-%m-%dT%H-%M') + suffix)
    assert not path.exists(), str(path)
    raw = ((json.dumps({**value, 'published_utc': now.isoformat()}, indent=2) + '\n').encode()
           if isinstance(value, dict) else value)
    assert b'\r' not in raw and raw.endswith(b'\n') and not raw.endswith(b'\n\n')
    assert all(line == line.rstrip() for line in raw.splitlines())
    path.write_bytes(raw)
    return path

def definitions(nodes, prefix=''):
    for node in nodes:
        if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
            name = prefix + node.name
            yield name, node
            yield from definitions(node.body, name + '.')
