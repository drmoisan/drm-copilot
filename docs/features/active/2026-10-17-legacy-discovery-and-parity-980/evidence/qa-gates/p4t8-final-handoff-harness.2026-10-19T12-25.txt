import ast
import base64
import datetime
import hashlib
import gzip
import json
from pathlib import Path
import re
import subprocess
import sys

root = Path.cwd()
scratch = Path(__file__).resolve().parent
qa = root / 'docs/features/active/2026-10-17-legacy-discovery-and-parity-980/evidence/qa-gates'
stamp = datetime.datetime.now(datetime.timezone.utc).strftime('%Y-%m-%dT%H-%M')
ordered_path = root / sys.argv[1]
ordered = json.loads(ordered_path.read_bytes())
candidate_path = scratch / ordered['candidate_manifest'].removeprefix('SCRATCH/')
candidate = json.loads(candidate_path.read_bytes())
qualification_path = root / ordered['unchanged_callable_qualification']['path']
assert hashlib.sha256(qualification_path.read_bytes()).hexdigest() == ordered['unchanged_callable_qualification']['sha256'] == '8b4a2467cddba5d0bd91bb35cf59994aeb6be344b3bb157833259da3fbed4edc'
qualification = json.loads(qualification_path.read_bytes())
unchanged_units = {(item['path'], item['kind'], item['name']): item for item in qualification['qualified_unchanged_units']}


def sha(content):
    return hashlib.sha256(content).hexdigest()


def ref(path):
    return {'path': path.relative_to(root).as_posix(), 'sha256': sha(path.read_bytes())}


def write(path, value):
    assert not path.exists(), path
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + '\n', encoding='utf-8', newline='\n')


assert sha(candidate_path.read_bytes()) == ordered['candidate_manifest_sha256']
assert [(item['stage'], item['label']) for item in ordered['receipts']] == [(1, 'format'), (2, 'lint'), (3, 'types'), (4, 'architecture'),
    (5, 'tests-optional-reverse'), (5, 'tests-compatibility-reverse'), (6, 'contracts'), (7, 'native')]
assert len(ordered['test_nodes']) == len(set(ordered['test_nodes'])) == sum(candidate['expected_test_counts'].values())
receipts = []
previous_finished = None
for entry in ordered['receipts']:
    path = root / entry['path']
    assert sha(path.read_bytes()) == entry['sha256']
    receipt = json.loads(path.read_bytes())
    assert receipt['exit_code'] == 0 and receipt['source_before'] == receipt['source_after'] == ordered['source_hashes']
    assert receipt['harness_hashes'] == ordered['harness_hashes']
    if previous_finished is not None:
        assert previous_finished <= receipt['started_at']
    previous_finished = receipt['finished_at']
    for stream in ('raw_stdout', 'raw_stderr'):
        raw = receipt[stream]
        assert sha((scratch / raw['path'].removeprefix('SCRATCH/')).read_bytes()) == raw['sha256']
    receipts.append({**entry, 'seconds': receipt['seconds'], 'markdown': ref(path.with_suffix('.md'))})
contracts = json.loads((root / ordered['receipts'][-2]['path']).read_bytes())
contract_count = int(re.findall(r'(\d+) passed', contracts['stdout'])[-1])
native = json.loads((root / ordered['native_path']).read_bytes())
assert len(native['checks']) == 22 and native['source_tree_unchanged'] and native['git_status_unchanged']
assert native['publication'] == 'not-attempted'
assert sha((scratch / ordered['raw_native']['path'].removeprefix('SCRATCH/')).read_bytes()) == ordered['raw_native']['sha256']
head = subprocess.run(['git', 'rev-parse', 'HEAD'], capture_output=True, check=True).stdout.decode().strip()
files, unit_ranges, coverage_refs = [], [], []
module_coverage = {}
for batch in ordered['coverage_batches']:
    path = root / batch['coverage_path']
    assert sha(path.read_bytes()) == batch['coverage_sha256']
    raw_coverage_bytes = (scratch / batch['raw_coverage']['path'].removeprefix('SCRATCH/')).read_bytes()
    assert sha(raw_coverage_bytes) == batch['raw_coverage']['sha256']
    raw_coverage_path = qa / ('p4t8-final-' + batch['batch'] + '-original-coverage.' + stamp + '.json')
    assert not raw_coverage_path.exists()
    raw_coverage_path.write_bytes(raw_coverage_bytes)
    coverage = json.loads(path.read_bytes())
    coverage_refs.append({'batch': batch['batch'], **ref(path), 'totals': coverage['totals'], 'raw': batch['raw_coverage'],
                          'test_count': len(batch['test_nodes']), 'test_module_order': batch['test_module_order'], 'canonical_original': ref(raw_coverage_path)})
    for relative, values in coverage['files'].items():
        assert relative not in module_coverage
        module_coverage[relative] = values['summary']
        tree = ast.parse((root / relative).read_bytes())
        def collect(nodes, prefix=''):
            for node in nodes:
                if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
                    family = 'classes' if isinstance(node, ast.ClassDef) else 'functions'
                    qualified = prefix + node.name
                    measured = values[family][qualified]['summary']
                    key = (relative, family, qualified)
                    status = 'new-or-changed-or-covered-unit'
                    if key in unchanged_units:
                        bound = unchanged_units[key]
                        assert sha((root / relative).read_bytes()) == qualification['source']['current_sha256']
                        assert [node.lineno, node.end_lineno] == bound['current_range']
                        assert measured['num_statements'] == bound['baseline_summary']['num_statements']
                        assert measured['percent_statements_covered'] >= bound['baseline_summary']['percent_statements_covered']
                        assert set(values[family][qualified]['missing_lines']) <= set(bound['current_missing_lines'])
                        status = 'unchanged-executable-unit-baseline-gap-retained'
                    else:
                        assert not measured['num_statements'] or measured['percent_statements_covered'] >= 90
                    unit_ranges.append({'path': relative, 'name': qualified, 'kind': type(node).__name__,
                        'start': node.lineno, 'end': node.end_lineno, 'summary': measured, 'status': status})
                    collect(node.body, qualified + '.')
        collect(tree.body)
assert len(module_coverage) == 8
for entry in candidate['files']:

    index = subprocess.run(['git', 'show', ':' + relative], capture_output=True, check=True).stdout

    assert content.replace(b'\r\n', b'\n') == index == committed
    blob = subprocess.run(['git', 'hash-object', '--no-filters', '--stdin'], input=content, capture_output=True, check=True).stdout.decode().strip()
    filtered = subprocess.run(['git', 'hash-object', '--path=' + relative, '--stdin'], input=content, capture_output=True, check=True).stdout.decode().strip()
    attributes = subprocess.run(['git', 'check-attr', 'text', 'eol', '--', relative], capture_output=True, check=True).stdout.decode().strip()
    files.append({'path': relative, 'sha256': sha(content), 'lines': len(content.splitlines()), 'working_git_blob': blob,
        'filtered_git_blob': filtered, 'index_sha256': sha(index), 'commit_sha256': sha(committed), 'commit': head,
        'git_attributes': attributes, 'stream_relation': 'identical' if content == index else 'CRLF-to-LF-only normalization'})

    assert sha((root / entry['path']).read_bytes()) == entry['sha256']

    snapshot_inputs['SCRATCH/' + name] = sha((scratch / name).read_bytes())

    harnesses.append({'executed_path': relative, 'executed_sha256': digest,

    'unchanged_callable_qualification': ref(qualification_path),

changed = []
for relative in module_coverage:
    diff = subprocess.run(['git', 'diff', '--no-ext-diff', '--unified=0', candidate['accepted_baseline'], 'HEAD', '--', relative], capture_output=True, check=True).stdout.decode()
    line_numbers = set()
    for start, count in re.findall(r'^@@ -\d+(?:,\d+)? \+(\d+)(?:,(\d+))? @@', diff, flags=re.M):
        line_numbers.update(range(int(start), int(start) + (int(count) if count else 1)))
    measured = next(json.loads((root / batch['coverage_path']).read_bytes())['files'][relative] for batch in ordered['coverage_batches'] if relative in json.loads((root / batch['coverage_path']).read_bytes())['files'])
    executable = set(measured['executed_lines']) | set(measured['missing_lines'])
    selected = sorted(executable & line_numbers)
    missing = sorted(set(measured['missing_lines']) & line_numbers)
    assert not missing, (relative, missing)
    changed.append({'path': relative, 'changed_executable_lines': selected, 'missing_changed_lines': missing,
                    'changed_statement_count': len(selected), 'coverage_percent': 100 if selected else None})
document['changed_line_measurement'] = changed

        encoded = base64.b64encode(gzip.compress(raw, mtime=0)).decode('ascii')
