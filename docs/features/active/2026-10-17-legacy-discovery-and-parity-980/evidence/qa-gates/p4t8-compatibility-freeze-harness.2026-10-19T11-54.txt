"""Freeze already executed compatibility evidence without rerunning tests."""
import ast
import base64
import datetime
import hashlib
import json
from pathlib import Path
import subprocess

root = Path.cwd()
scratch = Path(__file__).resolve().parent
feature = root / 'docs/features/active/2026-10-17-legacy-discovery-and-parity-980'
evidence = feature / 'evidence'
now = datetime.datetime.now(datetime.timezone.utc)
stamp = now.strftime('%Y-%m-%dT%H-%M')
baseline_path = evidence / 'baseline/p4t8-shared-baseline.2026-10-19T11-23.json'
binding_path = evidence / 'qa-gates/p4t8-compatibility-focused4-binding.2026-10-19T11-49.json'
structure_path = evidence / 'qa-gates/p4t8-compatibility-structure.2026-10-19T11-48.json'
binding = json.loads(binding_path.read_bytes())
baseline = json.loads(baseline_path.read_bytes())
structure = json.loads(structure_path.read_bytes())
coverage = json.loads((root / binding['coverage']['path']).read_bytes())

def sha(content):
    return hashlib.sha256(content).hexdigest()

def ref(path):
    return {'path': path.relative_to(root).as_posix(), 'sha256': sha(path.read_bytes())}

def emit(kind, stem, data):
    path = evidence / kind / (stem + '.' + stamp + '.json')
    with path.open('xb') as stream:
        stream.write((json.dumps(data, indent=2, sort_keys=True) + '\n').encode('utf-8'))
    return path

def markdown(kind, stem, text):
    path = evidence / kind / (stem + '.' + stamp + '.md')
    with path.open('xb') as stream:
        stream.write(('\n'.join(line.rstrip() for line in text.splitlines())+'\n').encode('utf-8'))
    return path

def git(*args):
    return subprocess.run(['git', *args], capture_output=True, check=True).stdout

sources = []
for path, expected in binding['sources'].items():
    content = (root / path).read_bytes()
    assert sha(content) == expected, path
    index = subprocess.run(['git', 'show', ':' + path], capture_output=True)
    committed = subprocess.run(['git', 'show', baseline['head'] + ':' + path], capture_output=True)
    sources.append({'path': path, 'sha256': expected, 'lines': len(content.splitlines()),
                    'working_git_object': git('hash-object', '--no-filters', path).decode().strip(),
                    'filtered_git_object': git('hash-object', '--path=' + path, path).decode().strip(),
                    'index_sha256': sha(index.stdout) if index.returncode == 0 else None,
                    'baseline_commit_sha256': sha(committed.stdout) if committed.returncode == 0 else None,
                    'qualification': 'Working candidate; index and accepted commit remain the predecessor until parent stages.'})
for path, expected in binding['shared_sources'].items():
    assert sha((root / path).read_bytes()) == expected, path
assert not structure['issues']
assert all(item['executable_ast_equal'] for item in structure['cli_preservation'])
assert {item['path']: item['sha256'] for item in structure['scope']} == binding['sources']

receipts = [json.loads((root / item['path']).read_bytes()) for item in binding['receipts']]
assert [item['stage'] for item in receipts] == ['format', 'lint', 'types', 'tests']
assert all(item['exit_code'] == 0 for item in receipts)
assert all(item['source_before'] == item['source_after'] == binding['sources'] for item in receipts)
assert all(item['shared_before'] == item['shared_after'] == binding['shared_sources'] for item in receipts)
nodes = receipts[-1]['test_node_order']
assert len(nodes) == len(set(nodes)) == 166

unit_rows = []
module_rows = []
baseline_coverage = json.loads((root / baseline['coverage']['path']).read_bytes())
old_modules = {key.replace('\\', '/'): value for key, value in baseline_coverage['files'].items()}
for path, data in coverage['files'].items():
    tree = ast.parse((root / path).read_bytes())
    definitions = {}
    def visit(items, parent=''):
        for node in items:
            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                name = parent + node.name
                definitions[name] = {'start': node.lineno, 'end': node.end_lineno}
                visit(node.body, name + '.')
    visit(tree.body)
    module_rows.append({'path': path, 'summary': data['summary'], 'missing_lines': data['missing_lines'],
                        'baseline_summary': old_modules[path]['summary'],
                        'statement_percentage_delta': data['summary']['percent_statements_covered']-old_modules[path]['summary']['percent_statements_covered']})
    for kind in ('classes', 'functions'):
        for name, unit in data[kind].items():
            if not name:
                continue
            unchanged = path.endswith('/stack_cli.py') and name in ('_utc_now', '_default_schema_path', '_build_parser', '_run', 'main_dotnet')
            status = 'unchanged-executable-cli' if unchanged else 'changed-or-new'
            if not unit['summary']['num_statements']:
                status = 'protocol-only-no-executable-statements'
            elif not unchanged:
                assert unit['summary']['percent_statements_covered'] >= 90, (path, name)
            unit_rows.append({'path': path, 'kind': kind, 'name': name, 'range': definitions[name],
                              'status': status, 'summary': unit['summary'], 'missing_lines': unit['missing_lines'],
                              'executed_branches': unit.get('executed_branches', []), 'missing_branches': unit.get('missing_branches', [])})
coverage_path = emit('regression-testing', 'p4t8-compatibility-coverage-ranges',
                    {'task': 'P4-T8', 'timestamp': stamp, 'coverage': binding['coverage'],
                     'modules': module_rows, 'units': unit_rows,
                     'qualification': 'Three uncovered statements are unchanged _utc_now and _default_schema_path bodies; their executable AST and baseline gaps are retained. Relocated detector coverage belongs to the optional owner.'})

harnesses = []
for stem in ('focused', 'precheck', 'structure'):
    path = scratch / ('issue-980-compatibility-' + stem + '.py')
    destination = evidence / 'qa-gates' / ('p4t8-compatibility-' + stem + '-harness.' + stamp + '.txt')
    with destination.open('xb') as stream:
        stream.write(path.read_bytes())
    harnesses.append({'scratch_path': 'SCRATCH/' + path.name, **ref(destination)})

streams = []
historical_receipts = []
for kind in ('qa-gates', 'regression-testing'):
    for path in sorted((evidence / kind).glob('p4t8-compatibility-*.json')):
        data = json.loads(path.read_bytes())
        if 'stage' not in data or 'raw' not in data:
            continue
        historical_receipts.append(ref(path))
        assert data['harness']['sha256'] == harnesses[0]['sha256'], path
        for label, item in data['raw'].items():
            content = (scratch / Path(item['path']).name).read_bytes()
            assert sha(content) == item['sha256'], item['path']
            streams.append({'receipt': path.relative_to(root).as_posix(), 'stream': label,
                            'path': item['path'], 'sha256': item['sha256'], 'base64': base64.b64encode(content).decode('ascii')})
for name in ('p4t8-compatibility-focused3-coverage.raw.json', 'p4t8-compatibility-focused4-coverage.raw.json'):
    content = (scratch / name).read_bytes()
    streams.append({'path': 'SCRATCH/' + name, 'sha256': sha(content), 'base64': base64.b64encode(content).decode('ascii')})
raw_path = emit('qa-gates', 'p4t8-compatibility-lossless-streams',
                {'task': 'P4-T8', 'timestamp': stamp, 'encoding': 'base64 preserves original raw bytes, including host paths; rendered receipts are portable', 'streams': streams})
