def digest(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()

def git(*args: str) -> bytes:
    return subprocess.check_output(['git', *args])

def file_binding(path: str) -> dict[str, object]:
    raw = (ROOT/path).read_bytes()
    committed = git('show', 'HEAD:'+path)
    normalized = raw.replace(b'\r\n',b'\n')
    assert normalized == committed.replace(b'\r\n',b'\n'), path
    return {
        'path':path,'sha256':digest(raw),'git_blob':git('rev-parse','HEAD:'+path).decode().strip(),
        'committed_sha256':digest(committed),'normalized_sha256':digest(normalized),
        'normalization':'exact' if raw == committed else 'CRLF-to-LF only',
        'physical_lines':len(raw.splitlines()),
    }

def receipt(path: Path) -> dict[str, str]:
    return {'path':path.as_posix(),'sha256':digest(path.read_bytes())}
